   122→export function SqlTool() {
   123→  const tables = getSqlTables();

   126→  const [result, setResult] = React.useState<RunResult | null>(null);
   127→  const [error, setError] = React.useState<string | null>(null);
   128→  const [openTable, setOpenTable] = React.useState<string | null>("customers");
   129→  const [exOpen, setExOpen] = React.useState(false);
   130→  const [tplOpen, setTplOpen] = React.useState(false);
   131→  const [histOpen, setHistOpen] = React.useState(false);
   132→  const [history, setHistory] = React.useState<string[]>(() => {
   133→    try { return JSON.parse(localStorage.getItem("aaa-sql-history") ?? "[]"); } catch { return []; }
   134→  });
   135→  const [checked, setChecked] = React.useState<Record<number, "pass" | "fail" | null>>({});
   136→  const [showHint, setShowHint] = React.useState<Record<number, boolean>>({});
   137→  const [showSolution, setShowSolution] = React.useState<Record<number, boolean>>({});
   138→
   139→  const run = (query?: string) => {
   140→    const q = (query ?? sql).trim();
   141→    if (!q) return;
   142→    const t0 = performance.now();
   143→    try {
   144→      const res = runSql(q, tables);
   145→      setResult({ ...res, ms: Math.max(1, Math.round(performance.now() - t0)) });
   146→      setError(null);
   147→      setHistory((h) => {
   148→        const next = [q, ...h.filter((x) => x !== q)].slice(0, 25);
   149→        try { localStorage.setItem("aaa-sql-history", JSON.stringify(next)); } catch {}
   150→        return next;
   151→      });
   152→    } catch (e) {
   153→      setError(e instanceof Error ? e.message : String(e));
   154→      setResult(null);
   155→    }
   156→  };
   157→
   158→  const check = (ex: Exercise) => {
   159→    try {
   160→      const user = runSql(sql, tables);
   161→      const sol = runSql(ex.solution, tables);
   162→      const norm = (r: RunResult) =>
   163→        JSON.stringify([r.columns.map((c) => c.toLowerCase()), r.rows.map((row) => row.map((v) => (v === null ? "" : typeof v === "number" ? +Number(v).toFixed(4) : String(v))))]);
   164→      const ok = norm(user) === norm(sol);
   165→      setChecked((c) => ({ ...c, [ex.id]: ok ? "pass" : "fail" }));
   166→      if (ok && !checked[ex.id]) addXp(15);
   167→    } catch (e) {
   168→      setChecked((c) => ({ ...c, [ex.id]: "fail" }));
   169→      setError(e instanceof Error ? e.message : String(e));
   170→    }
   171→  };
   172→
   173→  const loadExercise = (ex: Exercise) => {
   174→    setSql(`-- Exercise ${ex.id}: ${ex.title}\n-- ${ex.prompt}\n\n`);
   175→    setExOpen(false);
   176→  };
   177→
   178→  const exportResults = () => {
   179→    if (!result) return;
   180→    const csv = [
   181→      result.columns.map((c) => (/[,"]/.test(c) ? `"${c}"` : c)).join(","),
   182→      ...result.rows.map((r) => r.map((v) => {
   183→        const s = v === null ? "" : String(v);
   184→        return /[",]/.test(s) ? `"${s.replaceAll('"', '""')}"` : s;
   185→      }).join(",")),
   186→    ].join("\n");
   187→    downloadFile("query_results.csv", csv, "text/csv");
   188→  };
   189→
   190→  const passedCount = Object.values(checked).filter((v) => v === "pass").length;
   191→
   192→  /* dynamic tips */
   193→  const tips: string[] = React.useMemo(() => {
   194→    const t: string[] = [];
   195→    if (!result && !error) t.push("Hit **Run** (or Ctrl+↵) to execute the query. Start small: one table, a few columns, LIMIT 10.");
   196→    if (result) {
   197→      t.push(result.rows.length >= 500
   198→        ? "500+ rows returned — add a LIMIT or aggregate with GROUP BY so the answer is readable."
   199→        : `${result.rows.length} rows in ${result.ms} ms. Click **Export CSV** to reuse the result in Excel or a dashboard.`);
   200→    }
   201→    if (error) t.push("Read the error closely — SQL fails loudly, not silently. Check spelling, commas and the FROM table first.");
   202→    if (!Object.values(checked).some(Boolean)) t.push("The fastest way to learn: open **Exercises** and do #1, then #4. Check answer gives instant feedback.");
   203→    return t.slice(0, 3);
   204→  }, [result, error, checked]);

   219→              <BookOpenCheck className="h-4 w-4" /> Exercises {passedCount}/{EXERCISES.length}
   220→            </Button>
   221→          </>
   222→        }
   223→      />
   224→
   225→      <div className="grid grid-cols-1 gap-4 lg:grid-cols-[260px_1fr]">
   226→        {/* schema browser */}

   228→          <div className={PANEL_HEAD}><span className="flex items-center gap-2 text-xs font-semibold uppercase tracking-wide text-muted-foreground"><Database className="h-3.5 w-3.5 text-sky-500" /> Schema · store_db</span></div>

   232→                <button
   233→                  className={`flex w-full items-center justify-between px-3 py-2 text-left text-[13px] transition-colors ${openTable === t.name ? "bg-sky-500/10 text-sky-700 dark:text-sky-300" : "text-foreground/85 hover:bg-muted"}`}
   234→                  onClick={() => setOpenTable(openTable === t.name ? null : t.name)}

   238→                </button>

   289→            />
