    25→    25→const parseRef = (ref: string): [number, number] | null => {
    26→    26→  const m = /^\$?([A-Z])\$?(\d+)$/i.exec(ref.trim());
    27→    27→  if (!m) return null;

    36→    36→function evalSheetFormula(src: string, cells: Record<string, string>, stack: Set<string>): Val {
    37→    37→  let pos = 0;
    38→    38→  const ws = () => { while (pos < src.length && /\s/.test(src[pos])) pos++; };
    39→    39→
    40→    40→  function parseExpr(): Val { return parseCompare(); }
    41→    41→
    42→    42→  function parseCompare(): Val {
    43→    43→    let l = parseConcat();
    44→    44→    ws();
    45→    45→    const ops = ["<=", ">=", "<>", "=", "<", ">"];
    46→    46→    for (const op of ops) {
    47→    47→      if (src.startsWith(op, pos)) {
    48→    48→        pos += op.length;
    49→    49→        const r = parseConcat();
    50→    50→        const ln = typeof l === "number" ? l : parseFloat(String(l));
    51→    51→        const rn = typeof r === "number" ? r : parseFloat(String(r));
    52→    52→        const numeric = !isNaN(ln) && !isNaN(rn);
    53→    53→        switch (op) {
    54→    54→          case "=": return (numeric ? ln === rn : String(l) === String(r)) ? 1 : 0;
    55→    55→          case "<>": return (numeric ? ln !== rn : String(l) !== String(r)) ? 1 : 0;
    56→    56→          case "<": return numeric ? (ln < rn ? 1 : 0) : String(l) < String(r) ? 1 : 0;
    57→    57→          case ">": return numeric ? (ln > rn ? 1 : 0) : String(l) > String(r) ? 1 : 0;
    58→    58→          case "<=": return numeric ? (ln <= rn ? 1 : 0) : String(l) <= String(r) ? 1 : 0;
    59→    59→          case ">=": return numeric ? (ln >= rn ? 1 : 0) : String(l) >= String(r) ? 1 : 0;
    60→    60→        }
    61→    61→      }
    62→    62→    }
    63→    63→    return l;
    64→    64→  }
    65→    65→
    66→    66→  function parseConcat(): Val {
    67→    67→    let l = parseAdd();
    68→    68→    for (;;) {
    69→    69→      ws();
    70→    70→      if (src[pos] === "&") {
    71→    71→        pos++;
    72→    72→        const r = parseAdd();
    73→    73→        l = `${fmtRaw(l)}${fmtRaw(r)}`;
    74→    74→      } else return l;
    75→    75→    }
    76→    76→  }
    77→    77→
    78→    78→  function parseAdd(): Val {
    79→    79→    let l = parseMul();
    80→    80→    for (;;) {
    81→    81→      ws();
    82→    82→      if (src[pos] === "+") { pos++; l = num(l) + num(parseMul()); }
    83→    83→      else if (src[pos] === "-") { pos++; l = num(l) - num(parseMul()); }
    84→    84→      else return l;
    85→    85→    }
    86→    86→  }
    87→    87→  function parseMul(): Val {
    88→    88→    let l = parsePow();
    89→    89→    for (;;) {
    90→    90→      ws();
    91→    91→      if (src[pos] === "*") { pos++; l = num(l) * num(parsePow()); }
    92→    92→      else if (src[pos] === "/") {
    93→    93→        pos++;
    94→    94→        const d = num(parsePow());
    95→    95→        if (d === 0) throw new FormulaError("#DIV/0!");
    96→    96→        l = num(l) / d;
    97→    97→      } else return l;
    98→    98→    }
    99→    99→  }
   100→   100→  function parsePow(): Val {
   101→   101→    let l = parseUnary();
   102→   102→    for (;;) {
   103→   103→      ws();
   104→   104→      if (src[pos] === "^") { pos++; l = Math.pow(num(l), num(parseUnary())); }
   105→   105→      else return l;
   106→   106→    }
   107→   107→  }
   108→   108→  function parseUnary(): Val {
   109→   109→    ws();
   110→   110→    if (src[pos] === "-") { pos++; return -num(parseUnary()); }
   111→   111→    if (src[pos] === "+") { pos++; return num(parseUnary()); }
   112→   112→    return parsePrimary() as Val;
   113→   113→  }
   114→   114→
   115→   115→  function parsePrimary(): Val | Arg {
   116→   116→    ws();
   117→   117→    if (src[pos] === "(") {
   118→   118→      pos++;
   119→   119→      const v = parseExpr();
   120→   120→      ws();
   121→   121→      if (src[pos] !== ")") throw new FormulaError("#SYNTAX!");
   122→   122→      pos++;
   123→   123→      return v;
   124→   124→    }
   125→   125→    if (src[pos] === '"') {
   126→   126→      let s = "";
   127→   127→      pos++;
   128→   128→      while (pos < src.length && src[pos] !== '"') s += src[pos++];
   129→   129→      pos++;
   130→   130→      return s;
   131→   131→    }
   132→   132→    const numMatch = /^\d+(\.\d+)?/.exec(src.slice(pos));
   133→   133→    if (numMatch) { pos += numMatch[0].length; return parseFloat(numMatch[0]); }
   134→   134→    const fnMatch = /^([A-Z][A-Z0-9.]*)\(/i.exec(src.slice(pos));
   135→   135→    if (fnMatch) {
   136→   136→      const name = fnMatch[1].toUpperCase();
   137→   137→      pos += fnMatch[0].length;
   138→   138→      const args: Arg[] = [];
   139→   139→      if (src[pos] === ")") pos++;
   140→   140→      else {
   141→   141→        for (;;) {
   142→   142→          args.push(parseArg());
   143→   143→          if (src[pos] === ",") { pos++; continue; }
   144→   144→          if (src[pos] === ")") { pos++; break; }
   145→   145→          throw new FormulaError("#SYNTAX!");
   146→   146→        }
   147→   147→      }
   148→   148→      return callFn(name, args);
   149→   149→    }
   150→   150→    const refMatch = /^\$?[A-Za-z]\$?\d+/.exec(src.slice(pos));
   151→   151→    if (refMatch) {
   152→   152→      const ref = refMatch[0].toUpperCase();
   153→   153→      pos += refMatch[0].length;
   154→   154→      ws();
   155→   155→      if (src[pos] === ":") {
   156→   156→        pos++;
   157→   157→        const ref2Match = /^\$?[A-Za-z]\$?\d+/.exec(src.slice(pos));
   158→   158→        if (!ref2Match) throw new FormulaError("#SYNTAX!");
   159→   159→        pos += ref2Match[0].length;
   160→   160→        return rangeBox(ref, ref2Match[0].toUpperCase(), cells, stack);
   161→   161→      }
   162→   162→      return { kind: "value", v: cellValue(ref, cells, stack) };
   163→   163→    }
   164→   164→    if (/^TRUE$/i.test(src.slice(pos, pos + 4))) { pos += 4; return 1; }
   165→   165→    if (/^FALSE$/i.test(src.slice(pos, pos + 5))) { pos += 5; return 0; }
   166→   166→    throw new FormulaError("#NAME?");
   167→   167→  }
   168→   168→
   169→   169→  function parseArg(): Arg {
   170→   170→    ws();
   171→   171→    const startRef = /^\$?[A-Za-z]\$?\d+/.exec(src.slice(pos));
   172→   172→    if (startRef) {
   173→   173→      const ref = startRef[0].toUpperCase();
   174→   174→      const after = src.slice(pos + startRef[0].length);
   175→   175→      if (/^\s*:/.test(after)) {
   176→   176→        const m2 = /^\s*:\s*\$?[A-Za-z]\$?\d+/.exec(after);
   177→   177→        if (m2) {
   178→   178→          const ref2 = /^\s*:\s*(\$?[A-Za-z]\$?\d+)/.exec(after)![1].toUpperCase();
   179→   179→          pos += startRef[0].length + /^\s*:\s*\$?[A-Za-z]\$?\d+/.exec(after)![0].length;
   180→   180→          return rangeBox(ref, ref2, cells, stack);
   181→   181→        }
   182→   182→      }
   183→   183→    }
   184→   184→    const v = parseExpr();
   185→   185→    return { kind: "value", v };
   186→   186→  }
   187→   187→
   188→   188→  const v = parseExpr();
   189→   189→  ws();
   190→   190→  if (pos < src.length) throw new FormulaError("#SYNTAX!");
   191→   191→  if (typeof v === "object" && v !== null && "kind" in v) throw new FormulaError("#VALUE!");
   192→   192→  return v as Val;
   193→   193→}
   194→   194→
   195→   195→function fmtRaw(v: Val): string {
   196→   196→  return typeof v === "number" ? fmtNum(v) : String(v);
   197→   197→}
   198→   198→function num(v: Val): number {
   199→   199→  if (typeof v === "number") return v;
   200→   200→  const n = parseFloat(String(v ?? "").replace(/[$,%\s]/g, ""));
   201→   201→  if (isNaN(n)) throw new FormulaError("#VALUE!");
   202→   202→  return n;
   203→   203→}
   204→   204→function cellValue(ref: string, cells: Record<string, string>, stack: Set<string>): Val {
   205→   205→  if (stack.has(ref)) throw new FormulaError("#CYCLE!");
   206→   206→  const raw = cells[ref] ?? "";
   207→   207→  if (raw === "") return "";
   208→   208→  if (raw.startsWith("=")) {
   209→   209→    stack.add(ref);
   210→   210→    try {
   211→   211→      return evalSheetFormula(raw.slice(1), cells, stack);
   212→   212→    } finally {
   213→   213→      stack.delete(ref);
   214→   214→    }
   215→   215→  }
   216→   216→  const n = parseFloat(raw);
   217→   217→  return isNaN(n) || !/^-?[\d.,$%eE+-]+$/.test(raw) || /[\d]([.,]\d+)*[.,]\D/.test(raw) ? raw : n;
   218→   218→}
   219→   219→function rangeBox(a: string, b: string, cells: Record<string, string>, stack: Set<string>): Arg {
   220→   220→  const [c1, r1] = parseRef(a)!;
   221→   221→  const [c2, r2] = parseRef(b)!;
   222→   222→  const [cMin, cMax] = [Math.min(c1, c2), Math.max(c1, c2)];
   223→   223→  const [rMin, rMax] = [Math.min(r1, r2), Math.max(r1, r2)];
   224→   224→  const values: string[] = [];
   225→   225→  for (let r = rMin; r <= rMax; r++)
   226→   226→    for (let c = cMin; c <= cMax; c++)
   227→   227→      values.push(String(cellValue(refFor(c, r), cells, stack) ?? ""));
   228→   228→  return { kind: "range", values, w: cMax - cMin + 1, h: rMax - rMin + 1 };
   229→   229→}

   231→   231→function callFn(name: string, args: Arg[]): Val {
   232→   232→  const flat = (): string[] => {
   233→   233→    const out: string[] = [];
   234→   234→    for (const a of args) {
   235→   235→      if (a.kind === "range") out.push(...a.values);
   236→   236→      else out.push(String(a.v));
   237→   237→    }
   238→   238→    return out;
   239→   239→  };
   240→   240→  const numeric = (vals: string[]): number[] =>
   241→   241→    vals.map((v) => parseFloat(String(v ?? "").replace(/[$,\s]/g, ""))).filter((n) => !isNaN(n));
   242→   242→  const nonEmpty = (): string[] => flat().filter((v) => v !== "" && v !== "#CYCLE!");
   243→   243→  const scalar = (i: number): Val => (args[i]?.kind === "value" ? (args[i] as { v: Val }).v : num(args[i]?.values[0] ?? "0"));
   244→   244→  const text = (i: number): string => String(scalar(i) ?? "");
   245→   245→
   246→   246→  switch (name) {
   247→   247→    case "SUM": return numeric(flat()).reduce((s, n) => s + n, 0);
   248→   248→    case "AVERAGE": {
   249→   249→      const n = numeric(flat());
   250→   250→      return n.length ? n.reduce((s, x) => s + x, 0) / n.length : 0;
   251→   251→    }
   252→   252→    case "MEDIAN": {
   253→   253→      const n = numeric(flat()).sort((a, b) => a - b);
   254→   254→      if (!n.length) return 0;
   255→   255→      const mid = Math.floor(n.length / 2);
   256→   256→      return n.length % 2 ? n[mid] : (n[mid - 1] + n[mid]) / 2;
   257→   257→    }
   258→   258→    case "STDEV": {
   259→   259→      const n = numeric(flat());
   260→   260→      if (n.length < 2) return 0;
   261→   261→      const mean = n.reduce((s, x) => s + x, 0) / n.length;
   262→   262→      return Math.sqrt(n.reduce((s, x) => s + (x - mean) ** 2, 0) / (n.length - 1));
   263→   263→    }
   264→   264→    case "VAR": {
   265→   265→      const n = numeric(flat());
   266→   266→      if (n.length < 2) return 0;
   267→   267→      const mean = n.reduce((s, x) => s + x, 0) / n.length;
   268→   268→      return n.reduce((s, x) => s + (x - mean) ** 2, 0) / (n.length - 1);
   269→   269→    }
   270→   270→    case "COUNT": return numeric(flat()).length;
   271→   271→    case "COUNTA": return nonEmpty().length;
   272→   272→    case "COUNTBLANK": {
   273→   273→      const a = args[0];
   274→   274→      if (!a) return 0;
   275→   275→      const vals = a.kind === "range" ? a.values : [String(a.v)];
   276→   276→      return vals.filter((v) => v === "").length;
   277→   277→    }
   278→   278→    case "MIN": { const n = numeric(flat()); return n.length ? Math.min(...n) : 0; }
   279→   279→    case "MAX": { const n = numeric(flat()); return n.length ? Math.max(...n) : 0; }
   280→   280→    case "ROUND": {
   281→   281→      const d = args.length > 1 ? num(scalar(1)) : 0;
   282→   282→      return +num(scalar(0)).toFixed(d);
   283→   283→    }
   284→   284→    case "INT": return Math.floor(num(scalar(0)));
   285→   285→    case "MOD": return num(scalar(0)) % num(scalar(1));
   286→   286→    case "SQRT": return Math.sqrt(Math.abs(num(scalar(0))));
   287→   287→    case "POWER": return Math.pow(num(scalar(0)), num(scalar(1)));
   288→   288→    case "ABS": return Math.abs(num(scalar(0)));
   289→   289→    case "CONCAT":
   290→   290→    case "CONCATENATE": {
   291→   291→      let s = "";
   292→   292→      for (const a of args) {
   293→   293→        if (a.kind === "range") s += a.values.filter((v) => v !== "").join("");
   294→   294→        else s += fmtRaw(a.v);
   295→   295→      }
   296→   296→      return s;
   297→   297→    }
   298→   298→    case "VLOOKUP": {
   299→   299→      const lookup = scalar(0);
   300→   300→      const rng = args[1];
   301→   301→      if (!rng || rng.kind !== "range") throw new FormulaError("#VALUE!");
   302→   302→      const colIdx = Math.round(num(scalar(2)));
   303→   303→      if (colIdx < 1 || colIdx > rng.w) throw new FormulaError("#REF!");
   304→   304→      const lk = typeof lookup === "number" ? lookup : String(lookup).trim().toLowerCase();
   305→   305→      for (let r = 0; r < rng.h; r++) {
   306→   306→        const key = rng.values[r * rng.w];
   307→   307→        const keyN = typeof lookup === "number" ? parseFloat(String(key).replace(/[$,\s]/g, "")) : NaN;
   308→   308→        const match = typeof lookup === "number"
   309→   309→          ? !isNaN(keyN) && keyN === lookup
   310→   310→          : String(key).trim().toLowerCase() === lk;
   311→   311→        if (match) {
   312→   312→          const out = rng.values[r * rng.w + (colIdx - 1)];
   313→   313→          const outN = parseFloat(String(out).replace(/[$,\s]/g, ""));
   314→   314→          return isNaN(outN) || out === "" ? out : outN;
   315→   315→        }
   316→   316→      }
   317→   317→      throw new FormulaError("#N/A");
   318→   318→    }
   319→   319→    case "IF": {
   320→   320→      const cond = scalar(0);
   321→   321→      const yes = args[1]?.kind === "value" ? args[1].v : args[1]?.values[0] ?? "";
   322→   322→      const no = args[2]?.kind === "value" ? args[2].v : args[2]?.values[0] ?? "";
   323→   323→      const truthy = typeof cond === "number" ? cond !== 0 : String(cond).toUpperCase() === "TRUE" || (String(cond) !== "" && String(cond) !== "FALSE");
   324→   324→      return truthy ? yes : no;
   325→   325→    }
   326→   326→    case "AND": return args.every((a) => (a.kind === "value" ? typeof a.v === "number" ? a.v !== 0 : String(a.v).toUpperCase() !== "FALSE" && a.v !== "" : a.values.some((x) => x !== ""))) ? 1 : 0;
   327→   327→    case "OR": return args.some((a) => (a.kind === "value" ? typeof a.v === "number" ? a.v !== 0 : String(a.v).toUpperCase() === "TRUE" : a.values.some((x) => x !== ""))) ? 1 : 0;
   328→   328→    case "NOT": return num(scalar(0)) === 0 ? 1 : 0;
   329→   329→    case "UPPER": return text(0).toUpperCase();
   330→   330→    case "LOWER": return text(0).toLowerCase();
   331→   331→    case "PROPER": return text(0).toLowerCase().replace(/\b\w/g, (ch) => ch.toUpperCase());
   332→   332→    case "TRIM": return text(0).trim();
   333→   333→    case "LEN": return text(0).length;
   334→   334→    case "LEFT": return text(0).slice(0, args.length > 1 ? Math.round(num(scalar(1))) : 1);
   335→   335→    case "RIGHT": { const n = args.length > 1 ? Math.round(num(scalar(1))) : 1; return text(0).slice(-n || undefined); }
   336→   336→    case "MID": return text(0).slice(Math.round(num(scalar(1))) - 1, Math.round(num(scalar(1))) - 1 + Math.round(num(scalar(2))));
   337→   337→    case "VALUE": return num(scalar(0));
   338→   338→    case "TEXTJOIN": {
   339→   339→      const sep = text(0);
   340→   340→      const parts: string[] = [];
   341→   341→      for (const a of args.slice(2)) {
   342→   342→        if (a.kind === "range") parts.push(...a.values.filter((v) => v !== ""));
   343→   343→        else if (String(a.v) !== "") parts.push(String(a.v));
   344→   344→      }
   345→   345→      return parts.join(sep);
   346→   346→    }
   347→   347→    case "SUMIF":
   348→   348→    case "COUNTIF": {
   349→   349→      const rangeArg = args[0];
   350→   350→      const critArg = args[1];
   351→   351→      const crit = critArg.kind === "value" ? critArg.v : critArg.values[0] ?? "";
   352→   352→      const sumRange = name === "SUMIF" && args[2] ? args[2] : null;
   353→   353→      const test = (v: string): boolean => {
   354→   354→        const cs = String(crit);
   355→   355→        const m = /^(>=|<=|>|<|=|<>)(.*)$/.exec(cs);
   356→   356→        if (m) {
   357→   357→          const target = parseFloat(m[2].replace(/[$,\s]/g, ""));
   358→   358→          const value = parseFloat(String(v).replace(/[$,\s]/g, ""));
   359→   359→          if (!isNaN(target) && !isNaN(value)) {
   360→   360→            switch (m[1]) {
   361→   361→              case ">": return value > target;
   362→   362→              case "<": return value < target;
   363→   363→              case ">=": return value >= target;
   364→   364→              case "<=": return value <= target;
   365→   365→              case "=": return value === target;
   366→   366→              case "<>": return value !== target;
   367→   367→            }
   368→   368→          }
   369→   369→          return m[1] === "=" ? String(v) === m[2] : false;
   370→   370→        }
   371→   371→        return String(v).trim().toLowerCase() === cs.trim().toLowerCase();
   372→   372→      };
   373→   373→      if (name === "COUNTIF") {
   374→   374→        const vals = rangeArg.kind === "range" ? rangeArg.values : [String(rangeArg.v)];
   375→   375→        let c = 0;
   376→   376→        for (const v of vals) if (v !== "" && test(v)) c++;
   377→   377→        return c;
   378→   378→      }
   379→   379→      const vals = rangeArg.kind === "range" ? rangeArg.values : [String(rangeArg.v)];
   380→   380→      const sums = sumRange ? (sumRange.kind === "range" ? sumRange.values : [String(sumRange.v)]) : vals;
   381→   381→      let s = 0;
   382→   382→      vals.forEach((v, i) => {
   383→   383→        if (v !== "" && test(v)) {
   384→   384→          const n = parseFloat(String(sums[i] ?? "").replace(/[$,\s]/g, ""));
   385→   385→          if (!isNaN(n)) s += n;
   386→   386→        }
   387→   387→      });
   388→   388→      return s;
   389→   389→    }
   390→   390→    default: throw new FormulaError(`#${name}?`);
   391→   391→  }
   392→   392→}

   394→   394→function displayValue(ref: string, cells: Record<string, string>): string {
   395→   395→  const raw = cells[ref];
   396→   396→  if (raw === undefined || raw === "") return "";
   397→   397→  if (!raw.startsWith("=")) return raw;
   398→   398→  try {
   399→   399→    const v = evalSheetFormula(raw.slice(1), cells, new Set([ref]));
   400→   400→    if (typeof v === "number") {
   401→   401→      if (Number.isInteger(v)) return v.toLocaleString("en-US");
   402→   402→      return parseFloat(v.toFixed(4)).toLocaleString("en-US", { maximumFractionDigits: 4 });
   403→   403→    }
   404→   404→    return v;
   405→   405→  } catch (e) {
   406→   406→    return e instanceof FormulaError ? e.message : "#ERROR!";
   407→   407→  }
   408→   408→}
   409→   409→
   410→   410→function usedRange(cells: Record<string, string>): { rows: number; cols: number } {
   411→   411→  let maxR = 0, maxC = 0;
   412→   412→  for (const ref of Object.keys(cells)) {
   413→   413→    if (!cells[ref]) continue;
   414→   414→    const p = parseRef(ref);
   415→   415→    if (p) { maxR = Math.max(maxR, p[1]); maxC = Math.max(maxC, p[0]); }
   416→   416→  }
   417→   417→  return { rows: Math.min(maxR + 1, ROWS), cols: Math.min(maxC + 1, COLS) };
   418→   418→}
