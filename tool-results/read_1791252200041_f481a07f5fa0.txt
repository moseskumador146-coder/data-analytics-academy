    28→    28→const parseRef = (ref: string): [number, number] | null => {
    29→    29→  const m = /^\$?([A-Z])\$?(\d+)$/i.exec(ref.trim());
    30→    30→  if (!m) return null;

    39→    39→function evalSheetFormula(src: string, cells: Record<string, string>, stack: Set<string>): Val {
    40→    40→  let pos = 0;
    41→    41→  const ws = () => { while (pos < src.length && /\s/.test(src[pos])) pos++; };
    42→    42→
    43→    43→  function parseExpr(): Val { return parseCompare(); }
    44→    44→
    45→    45→  function parseCompare(): Val {
    46→    46→    let l = parseConcat();
    47→    47→    ws();
    48→    48→    const ops = ["<=", ">=", "<>", "=", "<", ">"];
    49→    49→    for (const op of ops) {
    50→    50→      if (src.startsWith(op, pos)) {
    51→    51→        pos += op.length;
    52→    52→        const r = parseConcat();
    53→    53→        const ln = typeof l === "number" ? l : parseFloat(String(l));
    54→    54→        const rn = typeof r === "number" ? r : parseFloat(String(r));
    55→    55→        const numeric = !isNaN(ln) && !isNaN(rn);
    56→    56→        switch (op) {
    57→    57→          case "=": return (numeric ? ln === rn : String(l) === String(r)) ? 1 : 0;
    58→    58→          case "<>": return (numeric ? ln !== rn : String(l) !== String(r)) ? 1 : 0;
    59→    59→          case "<": return numeric ? (ln < rn ? 1 : 0) : String(l) < String(r) ? 1 : 0;
    60→    60→          case ">": return numeric ? (ln > rn ? 1 : 0) : String(l) > String(r) ? 1 : 0;
    61→    61→          case "<=": return numeric ? (ln <= rn ? 1 : 0) : String(l) <= String(r) ? 1 : 0;
    62→    62→          case ">=": return numeric ? (ln >= rn ? 1 : 0) : String(l) >= String(r) ? 1 : 0;
    63→    63→        }
    64→    64→      }
    65→    65→    }
    66→    66→    return l;
    67→    67→  }
    68→    68→
    69→    69→  function parseConcat(): Val {
    70→    70→    let l = parseAdd();
    71→    71→    for (;;) {
    72→    72→      ws();
    73→    73→      if (src[pos] === "&") {
    74→    74→        pos++;
    75→    75→        const r = parseAdd();
    76→    76→        l = `${fmtRaw(l)}${fmtRaw(r)}`;
    77→    77→      } else return l;
    78→    78→    }
    79→    79→  }
    80→    80→
    81→    81→  function parseAdd(): Val {
    82→    82→    let l = parseMul();
    83→    83→    for (;;) {
    84→    84→      ws();
    85→    85→      if (src[pos] === "+") { pos++; l = num(l) + num(parseMul()); }
    86→    86→      else if (src[pos] === "-") { pos++; l = num(l) - num(parseMul()); }
    87→    87→      else return l;
    88→    88→    }
    89→    89→  }
    90→    90→  function parseMul(): Val {
    91→    91→    let l = parsePow();
    92→    92→    for (;;) {
    93→    93→      ws();
    94→    94→      if (src[pos] === "*") { pos++; l = num(l) * num(parsePow()); }
    95→    95→      else if (src[pos] === "/") {
    96→    96→        pos++;
    97→    97→        const d = num(parsePow());
    98→    98→        if (d === 0) throw new FormulaError("#DIV/0!");
    99→    99→        l = num(l) / d;
   100→   100→      } else return l;
   101→   101→    }
   102→   102→  }
   103→   103→  function parsePow(): Val {
   104→   104→    let l = parseUnary();
   105→   105→    for (;;) {
   106→   106→      ws();
   107→   107→      if (src[pos] === "^") { pos++; l = Math.pow(num(l), num(parseUnary())); }
   108→   108→      else return l;
   109→   109→    }
   110→   110→  }
   111→   111→  function parseUnary(): Val {
   112→   112→    ws();
   113→   113→    if (src[pos] === "-") { pos++; return -num(parseUnary()); }
   114→   114→    if (src[pos] === "+") { pos++; return num(parseUnary()); }
   115→   115→    return parsePrimary();
   116→   116→  }
   117→   117→
   118→   118→  function parsePrimary(): Val | Arg {
   119→   119→    ws();
   120→   120→    if (src[pos] === "(") {
   121→   121→      pos++;
   122→   122→      const v = parseExpr();
   123→   123→      ws();
   124→   124→      if (src[pos] !== ")") throw new FormulaError("#SYNTAX!");
   125→   125→      pos++;
   126→   126→      return v;
   127→   127→    }
   128→   128→    if (src[pos] === '"') {
   129→   129→      let s = "";
   130→   130→      pos++;
   131→   131→      while (pos < src.length && src[pos] !== '"') s += src[pos++];
   132→   132→      pos++;
   133→   133→      return s;
   134→   134→    }
   135→   135→    const numMatch = /^\d+(\.\d+)?/.exec(src.slice(pos));
   136→   136→    if (numMatch) { pos += numMatch[0].length; return parseFloat(numMatch[0]); }
   137→   137→    const fnMatch = /^([A-Z][A-Z0-9.]*)\(/i.exec(src.slice(pos));
   138→   138→    if (fnMatch) {
   139→   139→      const name = fnMatch[1].toUpperCase();
   140→   140→      pos += fnMatch[0].length;
   141→   141→      const args: Arg[] = [];
   142→   142→      if (src[pos] === ")") pos++;
   143→   143→      else {
   144→   144→        for (;;) {
   145→   145→          args.push(parseArg());
   146→   146→          if (src[pos] === ",") { pos++; continue; }
   147→   147→          if (src[pos] === ")") { pos++; break; }
   148→   148→          throw new FormulaError("#SYNTAX!");
   149→   149→        }
   150→   150→      }
   151→   151→      return callFn(name, args);
   152→   152→    }
   153→   153→    const refMatch = /^\$?[A-Za-z]\$?\d+/.exec(src.slice(pos));
   154→   154→    if (refMatch) {
   155→   155→      const ref = refMatch[0].toUpperCase();
   156→   156→      pos += refMatch[0].length;
   157→   157→      ws();
   158→   158→      if (src[pos] === ":") {
   159→   159→        pos++;
   160→   160→        const ref2Match = /^\$?[A-Za-z]\$?\d+/.exec(src.slice(pos));
   161→   161→        if (!ref2Match) throw new FormulaError("#SYNTAX!");
   162→   162→        pos += ref2Match[0].length;
   163→   163→        return { kind: "range", values: rangeValues(ref, ref2Match[0].toUpperCase(), cells, stack) };
   164→   164→      }
   165→   165→      return { kind: "value", v: cellValue(ref, cells, stack) };
   166→   166→    }
   167→   167→    if (/^TRUE$/i.test(src.slice(pos, pos + 4))) { pos += 4; return 1; }
   168→   168→    if (/^FALSE$/i.test(src.slice(pos, pos + 5))) { pos += 5; return 0; }
   169→   169→    throw new FormulaError("#NAME?");
   170→   170→  }
   171→   171→
   172→   172→  function parseArg(): Arg {
   173→   173→    ws();
   174→   174→    const startRef = /^\$?[A-Za-z]\$?\d+/.exec(src.slice(pos));
   175→   175→    if (startRef) {
   176→   176→      const ref = startRef[0].toUpperCase();
   177→   177→      const after = src.slice(pos + startRef[0].length);
   178→   178→      if (/^\s*:/.test(after)) {
   179→   179→        const m2 = /^\s*:\s*\$?[A-Za-z]\$?\d+/.exec(after);
   180→   180→        if (m2) {
   181→   181→          const ref2 = /^\s*:\s*(\$?[A-Za-z]\$?\d+)/.exec(after)![1].toUpperCase();
   182→   182→          pos += startRef[0].length + /^\s*:\s*\$?[A-Za-z]\$?\d+/.exec(after)![0].length;
   183→   183→          return { kind: "range", values: rangeValues(ref, ref2, cells, stack) };
   184→   184→        }
   185→   185→      }
   186→   186→    }
   187→   187→    const v = parseExpr();
   188→   188→    return { kind: "value", v };
   189→   189→  }
   190→   190→
   191→   191→  const v = parseExpr();
   192→   192→  ws();
   193→   193→  if (pos < src.length) throw new FormulaError("#SYNTAX!");
   194→   194→  return v;
   195→   195→}

   197→   197→function fmtRaw(v: Val): string {
   198→   198→  return typeof v === "number" ? fmtNum(v) : String(v);
   199→   199→}
   200→   200→function num(v: Val): number {
   201→   201→  if (typeof v === "number") return v;
   202→   202→  const n = parseFloat(String(v ?? "").replace(/[$,%\s]/g, ""));
   203→   203→  if (isNaN(n)) throw new FormulaError("#VALUE!");
   204→   204→  return n;
   205→   205→}
   206→   206→function cellValue(ref: string, cells: Record<string, string>, stack: Set<string>): Val {
   207→   207→  if (stack.has(ref)) throw new FormulaError("#CYCLE!");
   208→   208→  const raw = cells[ref] ?? "";
   209→   209→  if (raw === "") return "";
   210→   210→  if (raw.startsWith("=")) {
   211→   211→    stack.add(ref);
   212→   212→    try {
   213→   213→      return evalSheetFormula(raw.slice(1), cells, stack);
   214→   214→    } finally {
   215→   215→      stack.delete(ref);
   216→   216→    }
   217→   217→  }
   218→   218→  const n = parseFloat(raw);
   219→   219→  return isNaN(n) || !/^-?[\d.,$%eE+-]+$/.test(raw) || /[\d]([.,]\d+)*[.,]\D/.test(raw) ? raw : n;
   220→   220→}
   221→   221→function rangeValues(a: string, b: string, cells: Record<string, string>, stack: Set<string>): string[] {
   222→   222→  const [c1, r1] = parseRef(a)!;
   223→   223→  const [c2, r2] = parseRef(b)!;
   224→   224→  const [cMin, cMax] = [Math.min(c1, c2), Math.max(c1, c2)];
   225→   225→  const [rMin, rMax] = [Math.min(r1, r2), Math.max(r1, r2)];
   226→   226→  const out: string[] = [];
   227→   227→  for (let r = rMin; r <= rMax; r++)
   228→   228→    for (let c = cMin; c <= cMax; c++)
   229→   229→      out.push(String(cellValue(refFor(c, r), cells, stack) ?? ""));
   230→   230→  return out;
   231→   231→}

   233→   233→function callFn(name: string, args: Arg[], ): Val {
   234→   234→  const flat = (): string[] => {
   235→   235→    const out: string[] = [];
   236→   236→    for (const a of args) {
   237→   237→      if (a.kind === "range") out.push(...a.values);
   238→   238→      else out.push(String(a.v));
   239→   239→    }
   240→   240→    return out;
   241→   241→  };
   242→   242→  const numeric = (vals: string[]): number[] =>
   243→   243→    vals.map((v) => parseFloat(String(v ?? "").replace(/[$,\s]/g, ""))).filter((n) => !isNaN(n));
   244→   244→  const nonEmpty = (): string[] => flat().filter((v) => v !== "" && v !== "#CYCLE!");
   245→   245→
   246→   246→  switch (name) {
   247→   247→    case "SUM": return numeric(flat()).reduce((s, n) => s + n, 0);
   248→   248→    case "AVERAGE": {
   249→   249→      const n = numeric(flat());
   250→   250→      return n.length ? n.reduce((s, x) => s + x, 0) / n.length : 0;
   251→   251→    }
   252→   252→    case "MEDIAN": {
   253→   253→      const n = numeric(flat()).sort((a, b) => a - b);
   254→   254→      if (!n.length) return 0;
   255→   255→      const mid = Math.floor(n.length / 2);
   256→   256→      return n.length % 2 ? n[mid] : (n[mid - 1] + n[mid]) / 2;
   257→   257→    }
   258→   258→    case "STDEV": {
   259→   259→      const n = numeric(flat());
   260→   260→      if (n.length < 2) return 0;
   261→   261→      const mean = n.reduce((s, x) => s + x, 0) / n.length;
   262→   262→      return Math.sqrt(n.reduce((s, x) => s + (x - mean) ** 2, 0) / (n.length - 1));
   263→   263→    }
   264→   264→    case "COUNT": return numeric(flat()).length;
   265→   265→    case "COUNTA": return nonEmpty().length;
   266→   266→    case "MIN": { const n = numeric(flat()); return n.length ? Math.min(...n) : 0; }
   267→   267→    case "MAX": { const n = numeric(flat()); return n.length ? Math.max(...n) : 0; }
   268→   268→    case "ROUND": {
   269→   269→      const a = args.map((x) => (x.kind === "value" ? x.v : num(x.values[0] ?? "0")));
   270→   270→      const d = a.length > 1 ? num(a[1]) : 0;
   271→   271→      return +num(a[0]).toFixed(d);
   272→   272→    }
   273→   273→    case "ABS": return Math.abs(num(args[0].kind === "value" ? args[0].v : num(args[0].values[0] ?? "0")));
   274→   274→    case "IF": {
   275→   275→      const cond = args[0].kind === "value" ? args[0].v : num(args[0].values[0] ?? "0");
   276→   276→      const yes = args[1]?.kind === "value" ? args[1].v : args[1]?.values[0] ?? "";
   277→   277→      const no = args[2]?.kind === "value" ? args[2].v : args[2]?.values[0] ?? "";
   278→   278→      const truthy = typeof cond === "number" ? cond !== 0 : String(cond).toUpperCase() === "TRUE" || (String(cond) !== "" && String(cond) !== "FALSE");
   279→   279→      return truthy ? yes : no;
   280→   280→    }
   281→   281→    case "AND": return args.every((a) => (typeof a.v === "number" ? a.v !== 0 : String(a.v).toUpperCase() !== "FALSE" && a.v !== "")) ? 1 : 0;
   282→   282→    case "OR": return args.some((a) => (typeof a.v === "number" ? a.v !== 0 : String(a.v).toUpperCase() === "TRUE")) ? 1 : 0;
   283→   283→    case "UPPER": return String(args[0].kind === "value" ? args[0].v : args[0].values[0] ?? "").toUpperCase();
   284→   284→    case "LOWER": return String(args[0].kind === "value" ? args[0].v : args[0].values[0] ?? "").toLowerCase();
   285→   285→    case "TRIM": return String(args[0].kind === "value" ? args[0].v : args[0].values[0] ?? "").trim();
   286→   286→    case "LEN": return String(args[0].kind === "value" ? args[0].v : args[0].values[0] ?? "").length;
   287→   287→    case "SUMIF":
   288→   288→    case "COUNTIF": {
   289→   289→      const rangeArg = args[0];
   290→   290→      const critArg = args[1];
   291→   291→      const crit = critArg.kind === "value" ? critArg.v : critArg.values[0] ?? "";
   292→   292→      const sumRange = name === "SUMIF" && args[2] ? args[2] : null;
   293→   293→      const test = (v: string): boolean => {
   294→   294→        const cs = String(crit);
   295→   295→        const m = /^(>=|<=|>|<|=|<>)(.*)$/.exec(cs);
   296→   296→        if (m) {
   297→   297→          const target = parseFloat(m[2].replace(/[$,\s]/g, ""));
   298→   298→          const value = parseFloat(String(v).replace(/[$,\s]/g, ""));
   299→   299→          if (!isNaN(target) && !isNaN(value)) {
   300→   300→            switch (m[1]) {
   301→   301→              case ">": return value > target;
   302→   302→              case "<": return value < target;
   303→   303→              case ">=": return value >= target;
   304→   304→              case "<=": return value <= target;
   305→   305→              case "=": return value === target;
   306→   306→              case "<>": return value !== target;
   307→   307→            }
   308→   308→          }
   309→   309→          return m[1] === "=" ? String(v) === m[2] : false;
   310→   310→        }
   311→   311→        return String(v).trim().toLowerCase() === cs.trim().toLowerCase();
   312→   312→      };
   313→   313→      if (name === "COUNTIF") {
   314→   314→        const vals = rangeArg.kind === "range" ? rangeArg.values : [String(rangeArg.v)];
   315→   315→        let c = 0;
   316→   316→        for (const v of vals) if (v !== "" && test(v)) c++;
   317→   317→        return c;
   318→   318→      }
   319→   319→      const vals = rangeArg.kind === "range" ? rangeArg.values : [String(rangeArg.v)];
   320→   320→      const sums = sumRange ? (sumRange.kind === "range" ? sumRange.values : [String(sumRange.v)]) : vals;
   321→   321→      let s = 0;
   322→   322→      vals.forEach((v, i) => {
   323→   323→        if (v !== "" && test(v)) {
   324→   324→          const n = parseFloat(String(sums[i] ?? "").replace(/[$,\s]/g, ""));
   325→   325→          if (!isNaN(n)) s += n;
   326→   326→        }
   327→   327→      });
   328→   328→      return s;
   329→   329→    }
   330→   330→    default: throw new FormulaError(`#${name}?`);
   331→   331→  }
   332→   332→}

   334→   334→function displayValue(ref: string, cells: Record<string, string>): string {
   335→   335→  const raw = cells[ref];
   336→   336→  if (raw === undefined || raw === "") return "";
   337→   337→  if (!raw.startsWith("=")) return raw;
   338→   338→  try {
   339→   339→    const v = evalSheetFormula(raw.slice(1), cells, new Set([ref]));
   340→   340→    if (typeof v === "number") {
   341→   341→      if (Number.isInteger(v)) return v.toLocaleString("en-US");
   342→   342→      return parseFloat(v.toFixed(4)).toLocaleString("en-US", { maximumFractionDigits: 4 });
   343→   343→    }
   344→   344→    return v;
   345→   345→  } catch (e) {
   346→   346→    return e instanceof FormulaError ? e.message : "#ERROR!";
   347→   347→  }
   348→   348→}

   350→   350→function usedRange(cells: Record<string, string>): { rows: number; cols: number } {
   351→   351→  let maxR = 0, maxC = 0;
   352→   352→  for (const ref of Object.keys(cells)) {
   353→   353→    if (!cells[ref]) continue;
   354→   354→    const p = parseRef(ref);
   355→   355→    if (p) { maxR = Math.max(maxR, p[1]); maxC = Math.max(maxC, p[0]); }
   356→   356→  }
   357→   357→  return { rows: Math.min(maxR + 1, ROWS), cols: Math.min(maxC + 1, COLS) };
   358→   358→}
