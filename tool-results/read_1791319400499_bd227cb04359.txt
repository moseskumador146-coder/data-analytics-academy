    20→/* ================= formula engine ================= */
    21→const COLS = 26;
    22→const ROWS = 200;
    23→const colName = (i: number) => String.fromCharCode(65 + i);
    24→const refFor = (c: number, r: number) => `${colName(c)}${r + 1}`;
    25→const parseRef = (ref: string): [number, number] | null => {
    26→  const m = /^\$?([A-Z])\$?(\d+)$/i.exec(ref.trim());
    27→  if (!m) return null;
    28→  return [m[1].toUpperCase().charCodeAt(0) - 65, parseInt(m[2], 10) - 1];
    29→};

    31→type Val = string | number;
    32→type Arg = { kind: "value"; v: Val } | { kind: "range"; values: string[]; w: number; h: number };

    34→class FormulaError extends Error {}
    35→
    36→function evalSheetFormula(src: string, cells: Record<string, string>, stack: Set<string>): Val {
    37→  let pos = 0;
    38→  const ws = () => { while (pos < src.length && /\s/.test(src[pos])) pos++; };
    39→
    40→  function parseExpr(): Val { return parseCompare(); }
    41→
    42→  function parseCompare(): Val {
    43→    let l = parseConcat();
    44→    ws();
    45→    const ops = ["<=", ">=", "<>", "=", "<", ">"];
    46→    for (const op of ops) {
    47→      if (src.startsWith(op, pos)) {
    48→        pos += op.length;
    49→        const r = parseConcat();
    50→        const ln = typeof l === "number" ? l : parseFloat(String(l));
    51→        const rn = typeof r === "number" ? r : parseFloat(String(r));
    52→        const numeric = !isNaN(ln) && !isNaN(rn);
    53→        switch (op) {
    54→          case "=": return (numeric ? ln === rn : String(l) === String(r)) ? 1 : 0;
    55→          case "<>": return (numeric ? ln !== rn : String(l) !== String(r)) ? 1 : 0;
    56→          case "<": return numeric ? (ln < rn ? 1 : 0) : String(l) < String(r) ? 1 : 0;
    57→          case ">": return numeric ? (ln > rn ? 1 : 0) : String(l) > String(r) ? 1 : 0;
    58→          case "<=": return numeric ? (ln <= rn ? 1 : 0) : String(l) <= String(r) ? 1 : 0;
    59→          case ">=": return numeric ? (ln >= rn ? 1 : 0) : String(l) >= String(r) ? 1 : 0;
    60→        }
    61→      }
    62→    }
    63→    return l;
    64→  }
    65→
    66→  function parseConcat(): Val {
    67→    let l = parseAdd();
    68→    for (;;) {
    69→      ws();
    70→      if (src[pos] === "&") {
    71→        pos++;
    72→        const r = parseAdd();
    73→        l = `${fmtRaw(l)}${fmtRaw(r)}`;
    74→      } else return l;
    75→    }
    76→  }
    77→
    78→  function parseAdd(): Val {
    79→    let l = parseMul();
    80→    for (;;) {
    81→      ws();
    82→      if (src[pos] === "+") { pos++; l = num(l) + num(parseMul()); }
    83→      else if (src[pos] === "-") { pos++; l = num(l) - num(parseMul()); }
    84→      else return l;
    85→    }
    86→  }
    87→  function parseMul(): Val {
    88→    let l = parsePow();
    89→    for (;;) {
    90→      ws();
    91→      if (src[pos] === "*") { pos++; l = num(l) * num(parsePow()); }
    92→      else if (src[pos] === "/") {
    93→        pos++;
    94→        const d = num(parsePow());
    95→        if (d === 0) throw new FormulaError("#DIV/0!");
    96→        l = num(l) / d;
    97→      } else return l;
    98→    }
    99→  }
   100→  function parsePow(): Val {
   101→    let l = parseUnary();
   102→    for (;;) {
   103→      ws();
   104→      if (src[pos] === "^") { pos++; l = Math.pow(num(l), num(parseUnary())); }
   105→      else return l;
   106→    }
   107→  }
   108→  function parseUnary(): Val {
   109→    ws();
   110→    if (src[pos] === "-") { pos++; return -num(parseUnary()); }
   111→    if (src[pos] === "+") { pos++; return num(parseUnary()); }
   112→    return parsePrimary() as Val;
   113→  }
   114→
   115→  function parsePrimary(): Val | Arg {
   116→    ws();
   117→    if (src[pos] === "(") {
   118→      pos++;
   119→      const v = parseExpr();
   120→      ws();
   121→      if (src[pos] !== ")") throw new FormulaError("#SYNTAX!");
   122→      pos++;
   123→      return v;
   124→    }
   125→    if (src[pos] === '"') {
   126→      let s = "";
   127→      pos++;
   128→      while (pos < src.length && src[pos] !== '"') s += src[pos++];
   129→      pos++;
   130→      return s;
   131→    }
   132→    /* full-column refs: F:F · $F:$F  →  whole column range */
   133→    const colRange = /^\$?([A-Za-z])\$?\s*:\s*\$?([A-Za-z])\$?(?!\d)/.exec(src.slice(pos));
   134→    if (colRange) {
   135→      const c1 = colRange[1].toUpperCase().charCodeAt(0) - 65;
   136→      const c2 = colRange[2].toUpperCase().charCodeAt(0) - 65;
   137→      pos += colRange[0].length;
   138→      if (c1 >= COLS || c2 >= COLS) throw new FormulaError("#REF!");
   139→      return rangeBox(refFor(Math.min(c1, c2), 0), refFor(Math.max(c1, c2), ROWS - 1), cells, stack);
   140→    }
   141→    /* full-row refs: 2:4 → whole-row range */
   142→    const rowRange = /^\$?(\d+)\s*:\s*\$?(\d+)/.exec(src.slice(pos));
   143→    if (rowRange) {
   144→      const r1 = Math.max(0, Math.min(+rowRange[1], +rowRange[2]) - 1);
   145→      const r2 = Math.min(ROWS - 1, Math.max(+rowRange[1], +rowRange[2]) - 1);
   146→      pos += rowRange[0].length;
   147→      return rangeBox(refFor(0, r1), refFor(COLS - 1, r2), cells, stack);
   148→    }
   149→    const numMatch = /^\d+(\.\d+)?/.exec(src.slice(pos));
   150→    if (numMatch) { pos += numMatch[0].length; return parseFloat(numMatch[0]); }
   151→    const fnMatch = /^([A-Z][A-Z0-9.]*)\(/i.exec(src.slice(pos));
   152→    if (fnMatch) {
   153→      const name = fnMatch[1].toUpperCase();
   154→      pos += fnMatch[0].length;
   155→      if (name === "IFERROR" || name === "IFNA") return parseIfError(name);
   156→      const args: Arg[] = [];
   157→      if (src[pos] === ")") pos++;
   158→      else {
   159→        for (;;) {
   160→          args.push(parseArg());
   161→          if (src[pos] === ",") { pos++; continue; }
   162→          if (src[pos] === ")") { pos++; break; }
   163→          throw new FormulaError("#SYNTAX!");
   164→        }
   165→      }
   166→      return callFn(name, args);
   167→    }
   168→    const refMatch = /^\$?[A-Za-z]\$?\d+/.exec(src.slice(pos));
   169→    if (refMatch) {
   170→      const ref = refMatch[0].toUpperCase();
   171→      pos += refMatch[0].length;
   172→      ws();
   173→      if (src[pos] === ":") {
   174→        pos++;
   175→        const ref2Match = /^\$?[A-Za-z]\$?\d+/.exec(src.slice(pos));
   176→        if (!ref2Match) throw new FormulaError("#SYNTAX!");
   177→        pos += ref2Match[0].length;
   178→        return rangeBox(ref, ref2Match[0].toUpperCase(), cells, stack);
   179→      }
   180→      return { kind: "value", v: cellValue(ref, cells, stack) };
   181→    }
   182→    if (/^TRUE$/i.test(src.slice(pos, pos + 4))) { pos += 4; return 1; }
   183→    if (/^FALSE$/i.test(src.slice(pos, pos + 5))) { pos += 5; return 0; }
   184→    throw new FormulaError("#NAME?");
   185→  }
   186→
   187→  function parseArg(): Arg {
   188→    ws();
   189→    /* full-column / full-row refs */
   190→    const colRange = /^\$?([A-Za-z])\$?\s*:\s*\$?([A-Za-z])\$?(?!\d)/.exec(src.slice(pos));
   191→    if (colRange) {
   192→      const c1 = colRange[1].toUpperCase().charCodeAt(0) - 65;
   193→      const c2 = colRange[2].toUpperCase().charCodeAt(0) - 65;
   194→      pos += colRange[0].length;
   195→      if (c1 >= COLS || c2 >= COLS) throw new FormulaError("#REF!");
   196→      return rangeBox(refFor(Math.min(c1, c2), 0), refFor(Math.max(c1, c2), ROWS - 1), cells, stack);
   197→    }
   198→    const rowRange = /^\$?(\d+)\s*:\s*\$?(\d+)/.exec(src.slice(pos));
   199→    if (rowRange) {
   200→      const r1 = Math.max(0, Math.min(+rowRange[1], +rowRange[2]) - 1);
   201→      const r2 = Math.min(ROWS - 1, Math.max(+rowRange[1], +rowRange[2]) - 1);
   202→      pos += rowRange[0].length;
   203→      return rangeBox(refFor(0, r1), refFor(COLS - 1, r2), cells, stack);
   204→    }
   205→    const startRef = /^\$?[A-Za-z]\$?\d+/.exec(src.slice(pos));
   206→    if (startRef) {
   207→      const ref = startRef[0].toUpperCase();
   208→      const after = src.slice(pos + startRef[0].length);
   209→      if (/^\s*:/.test(after)) {
   210→        const m2 = /^\s*:\s*\$?[A-Za-z]\$?\d+/.exec(after);
   211→        if (m2) {
   212→          const ref2 = /^\s*:\s*(\$?[A-Za-z]\$?\d+)/.exec(after)![1].toUpperCase();
   213→          pos += startRef[0].length + /^\s*:\s*\$?[A-Za-z]\$?\d+/.exec(after)![0].length;
   214→          return rangeBox(ref, ref2, cells, stack);
   215→        }
   216→      }
   217→    }
   218→    const v = parseExpr();
   219→    return { kind: "value", v };
   220→  }
   221→
   222→  /* IFERROR / IFNA need lazy evaluation: the fallback must only be parsed when
   223→     the main argument actually throws. Implemented at the parser level. */
   224→  function argValue(a: Arg): Val {
   225→    return a.kind === "value" ? a.v : a.values[0] ?? "";
   226→  }
   227→  function skipToCommaOrClose(): void {
   228→    let depth = 0;
   229→    while (pos < src.length) {
   230→      const ch = src[pos];
   231→      if (ch === "(") depth++;
   232→      else if (ch === ")") { if (depth === 0) return; depth--; }
   233→      else if (ch === "," && depth === 0) return;
   234→      pos++;
   235→    }
   236→  }
   237→  function parseIfError(name: "IFERROR" | "IFNA"): Val {
   238→    ws();
   239→    const start = pos;
   240→    let main: Arg | null = null;
   241→    let failed: unknown = null;
   242→    try {
   243→      main = parseArg();
   244→    } catch (e) {
   245→      failed = e;
   246→      pos = start;
   247→      skipToCommaOrClose();
   248→    }
   249→    const isNa = failed instanceof FormulaError && failed.message === "#N/A";
   250→    if (failed && name === "IFNA" && !isNa) throw failed;
   251→    if (main) {
   252→      ws();
   253→      if (src[pos] === ",") {
   254→        pos++;
   255→        try { parseArg(); } catch { /* fallback errored — main value still wins */ skipToCommaOrClose(); }
   256→      }
   257→      ws();
   258→      if (src[pos] === ")") pos++;
   259→      return argValue(main);
   260→    }
   261→    ws();
   262→    if (src[pos] !== ",") throw failed; // no fallback → surface the error
   263→    pos++;
   264→    const fb = parseArg();
   265→    ws();
   266→    if (src[pos] === ")") pos++;
   267→    return argValue(fb);
   268→  }
   269→
   270→  const v = parseExpr();
   271→  ws();
   272→  if (pos < src.length) throw new FormulaError("#SYNTAX!");
   273→  if (typeof v === "object" && v !== null && "kind" in v) throw new FormulaError("#VALUE!");
   274→  return v as Val;
   275→}
   276→
   277→function fmtRaw(v: Val): string {
   278→  return typeof v === "number" ? fmtNum(v) : String(v);
   279→}
   280→function num(v: Val): number {
   281→  if (typeof v === "number") return v;
   282→  const n = parseFloat(String(v ?? "").replace(/[$,%\s]/g, ""));
   283→  if (isNaN(n)) throw new FormulaError("#VALUE!");
   284→  return n;
   285→}
   286→function cellValue(ref: string, cells: Record<string, string>, stack: Set<string>): Val {
   287→  if (stack.has(ref)) throw new FormulaError("#CYCLE!");
   288→  const raw = cells[ref] ?? "";
   289→  if (raw === "") return "";
   290→  if (raw.startsWith("=")) {
   291→    stack.add(ref);
   292→    try {
   293→      return evalSheetFormula(raw.slice(1), cells, stack);
   294→    } finally {
   295→      stack.delete(ref);
   296→    }
   297→  }
   298→  const n = parseFloat(raw);
   299→  return isNaN(n) || !/^-?[\d.,$%eE+-]+$/.test(raw) || /[\d]([.,]\d+)*[.,]\D/.test(raw) ? raw : n;
   300→}
   301→function rangeBox(a: string, b: string, cells: Record<string, string>, stack: Set<string>): Arg {
   302→  const [c1, r1] = parseRef(a)!;
   303→  const [c2, r2] = parseRef(b)!;
   304→  const [cMin, cMax] = [Math.min(c1, c2), Math.max(c1, c2)];
   305→  const [rMin, rMax] = [Math.min(r1, r2), Math.max(r1, r2)];
   306→  const values: string[] = [];
   307→  for (let r = rMin; r <= rMax; r++)
   308→    for (let c = cMin; c <= cMax; c++)
   309→      values.push(String(cellValue(refFor(c, r), cells, stack) ?? ""));
   310→  return { kind: "range", values, w: cMax - cMin + 1, h: rMax - rMin + 1 };
   311→}

   313→function callFn(name: string, args: Arg[]): Val {
   314→  const flat = (): string[] => {
   315→    const out: string[] = [];
   316→    for (const a of args) {
   317→      if (a.kind === "range") out.push(...a.values);
   318→      else out.push(String(a.v));
   319→    }
   320→    return out;
   321→  };
   322→  const numeric = (vals: string[]): number[] =>
   323→    vals.map((v) => parseFloat(String(v ?? "").replace(/[$,\s]/g, ""))).filter((n) => !isNaN(n));
   324→  const nonEmpty = (): string[] => flat().filter((v) => v !== "" && v !== "#CYCLE!");
   325→  const scalar = (i: number): Val => (args[i]?.kind === "value" ? (args[i] as { v: Val }).v : num(args[i]?.values[0] ?? "0"));
   326→  const text = (i: number): string => String(scalar(i) ?? "");
   327→
   328→  switch (name) {
   329→    case "SUM": return numeric(flat()).reduce((s, n) => s + n, 0);
   330→    case "AVERAGE": {
   331→      const n = numeric(flat());
   332→      return n.length ? n.reduce((s, x) => s + x, 0) / n.length : 0;
   333→    }
   334→    case "MEDIAN": {
   335→      const n = numeric(flat()).sort((a, b) => a - b);
   336→      if (!n.length) return 0;
   337→      const mid = Math.floor(n.length / 2);
   338→      return n.length % 2 ? n[mid] : (n[mid - 1] + n[mid]) / 2;
   339→    }
   340→    case "STDEV": {
   341→      const n = numeric(flat());
   342→      if (n.length < 2) return 0;
   343→      const mean = n.reduce((s, x) => s + x, 0) / n.length;
   344→      return Math.sqrt(n.reduce((s, x) => s + (x - mean) ** 2, 0) / (n.length - 1));
   345→    }
   346→    case "VAR": {
   347→      const n = numeric(flat());
   348→      if (n.length < 2) return 0;
   349→      const mean = n.reduce((s, x) => s + x, 0) / n.length;
   350→      return n.reduce((s, x) => s + (x - mean) ** 2, 0) / (n.length - 1);
   351→    }
   352→    case "COUNT": return numeric(flat()).length;
   353→    case "COUNTA": return nonEmpty().length;
   354→    case "COUNTBLANK": {
   355→      const a = args[0];
   356→      if (!a) return 0;
   357→      const vals = a.kind === "range" ? a.values : [String(a.v)];
   358→      return vals.filter((v) => v === "").length;
   359→    }
   360→    case "MIN": { const n = numeric(flat()); return n.length ? Math.min(...n) : 0; }
   361→    case "MAX": { const n = numeric(flat()); return n.length ? Math.max(...n) : 0; }
   362→    case "ROUND": {
   363→      const d = args.length > 1 ? num(scalar(1)) : 0;
   364→      return +num(scalar(0)).toFixed(d);
   365→    }
   366→    case "INT": return Math.floor(num(scalar(0)));
   367→    case "MOD": return num(scalar(0)) % num(scalar(1));
   368→    case "SQRT": return Math.sqrt(Math.abs(num(scalar(0))));
   369→    case "POWER": return Math.pow(num(scalar(0)), num(scalar(1)));
   370→    case "ABS": return Math.abs(num(scalar(0)));
   371→    case "CONCAT":
   372→    case "CONCATENATE": {
   373→      let s = "";
   374→      for (const a of args) {
   375→        if (a.kind === "range") s += a.values.filter((v) => v !== "").join("");
   376→        else s += fmtRaw(a.v);
   377→      }
   378→      return s;
   379→    }
   380→    case "VLOOKUP": {
   381→      const lookup = scalar(0);
   382→      const rng = args[1];
   383→      if (!rng || rng.kind !== "range") throw new FormulaError("#VALUE!");
   384→      const colIdx = Math.round(num(scalar(2)));
   385→      if (colIdx < 1 || colIdx > rng.w) throw new FormulaError("#REF!");
   386→      const lk = typeof lookup === "number" ? lookup : String(lookup).trim().toLowerCase();
   387→      for (let r = 0; r < rng.h; r++) {
   388→        const key = rng.values[r * rng.w];
   389→        const keyN = typeof lookup === "number" ? parseFloat(String(key).replace(/[$,\s]/g, "")) : NaN;
   390→        const match = typeof lookup === "number"
   391→          ? !isNaN(keyN) && keyN === lookup
   392→          : String(key).trim().toLowerCase() === lk;
   393→        if (match) {
   394→          const out = rng.values[r * rng.w + (colIdx - 1)];
   395→          const outN = parseFloat(String(out).replace(/[$,\s]/g, ""));
   396→          return isNaN(outN) || out === "" ? out : outN;
   397→        }
   398→      }
   399→      throw new FormulaError("#N/A");
   400→    }
   401→    case "IF": {
   402→      const cond = scalar(0);
   403→      const yes = args[1]?.kind === "value" ? args[1].v : args[1]?.values[0] ?? "";
   404→      const no = args[2]?.kind === "value" ? args[2].v : args[2]?.values[0] ?? "";
   405→      const truthy = typeof cond === "number" ? cond !== 0 : String(cond).toUpperCase() === "TRUE" || (String(cond) !== "" && String(cond) !== "FALSE");
   406→      return truthy ? yes : no;
   407→    }
   408→    case "AND": return args.every((a) => (a.kind === "value" ? typeof a.v === "number" ? a.v !== 0 : String(a.v).toUpperCase() !== "FALSE" && a.v !== "" : a.values.some((x) => x !== ""))) ? 1 : 0;
   409→    case "OR": return args.some((a) => (a.kind === "value" ? typeof a.v === "number" ? a.v !== 0 : String(a.v).toUpperCase() === "TRUE" : a.values.some((x) => x !== ""))) ? 1 : 0;
   410→    case "NOT": return num(scalar(0)) === 0 ? 1 : 0;
   411→    case "UPPER": return text(0).toUpperCase();
   412→    case "LOWER": return text(0).toLowerCase();
   413→    case "PROPER": return text(0).toLowerCase().replace(/\b\w/g, (ch) => ch.toUpperCase());
   414→    case "TRIM": return text(0).trim();
   415→    case "LEN": return text(0).length;
   416→    case "LEFT": return text(0).slice(0, args.length > 1 ? Math.round(num(scalar(1))) : 1);
   417→    case "RIGHT": { const n = args.length > 1 ? Math.round(num(scalar(1))) : 1; return text(0).slice(-n || undefined); }
   418→    case "MID": return text(0).slice(Math.round(num(scalar(1))) - 1, Math.round(num(scalar(1))) - 1 + Math.round(num(scalar(2))));
   419→    case "VALUE": return num(scalar(0));
   420→    case "TEXTJOIN": {
   421→      const sep = text(0);
   422→      const parts: string[] = [];
   423→      for (const a of args.slice(2)) {
   424→        if (a.kind === "range") parts.push(...a.values.filter((v) => v !== ""));
   425→        else if (String(a.v) !== "") parts.push(String(a.v));
   426→      }
   427→      return parts.join(sep);
   428→    }
   429→    case "SUMIF":
   430→    case "AVERAGEIF":
   431→    case "COUNTIF": {
   432→      const rangeArg = args[0];
   433→      const critArg = args[1];
   434→      const crit = critArg.kind === "value" ? critArg.v : critArg.values[0] ?? "";
   435→      const sumRange = name === "SUMIF" && args[2] ? args[2] : null;
   436→      const test = (v: string): boolean => {
   437→        const cs = String(crit);
   438→        const m = /^(>=|<=|>|<|=|<>)(.*)$/.exec(cs);
   439→        if (m) {
   440→          const target = parseFloat(m[2].replace(/[$,\s]/g, ""));
   441→          const value = parseFloat(String(v).replace(/[$,\s]/g, ""));
   442→          if (!isNaN(target) && !isNaN(value)) {
   443→            switch (m[1]) {
   444→              case ">": return value > target;
   445→              case "<": return value < target;
   446→              case ">=": return value >= target;
   447→              case "<=": return value <= target;
   448→              case "=": return value === target;
   449→              case "<>": return value !== target;
   450→            }
   451→          }
   452→          return m[1] === "=" ? String(v) === m[2] : false;
   453→        }
   454→        return String(v).trim().toLowerCase() === cs.trim().toLowerCase();
   455→      };
   456→      if (name === "COUNTIF") {
   457→        const vals = rangeArg.kind === "range" ? rangeArg.values : [String(rangeArg.v)];
   458→        let c = 0;
   459→        for (const v of vals) if (v !== "" && test(v)) c++;
   460→        return c;
   461→      }
   462→      const vals = rangeArg.kind === "range" ? rangeArg.values : [String(rangeArg.v)];
   463→      const sums = sumRange ? (sumRange.kind === "range" ? sumRange.values : [String(sumRange.v)]) : vals;
   464→      let s = 0;
   465→      let matched = 0;
   466→      vals.forEach((v, i) => {
   467→        if (v !== "" && test(v)) {
   468→          const n = parseFloat(String(sums[i] ?? "").replace(/[$,\s]/g, ""));
   469→          if (!isNaN(n)) { s += n; matched++; }
   470→        }
   471→      });
   472→      return name === "AVERAGEIF" ? (matched ? s / matched : 0) : s;
   473→    }
   474→    case "SUMIFS":
   475→    case "COUNTIFS": {
   476→      // SUMIFS(sum_range, crit_range1, crit1, ...) · COUNTIFS(crit_range1, crit1, ...)
   477→      const pairs: { vals: string[]; crit: string }[] = [];
   478→      let sumVals: string[] = [];
   479→      if (name === "SUMIFS") {
   480→        const sr = args[0];
   481→        sumVals = sr ? (sr.kind === "range" ? sr.values : [String(sr.v)]) : [];
   482→        for (let i = 1; i + 1 < args.length; i += 2) {
   483→          const r = args[i];
   484→          const c = args[i + 1];
   485→          pairs.push({ vals: r ? (r.kind === "range" ? r.values : [String(r.v)]) : [], crit: c ? (c.kind === "value" ? String(c.v) : String(c.values[0] ?? "")) : "" });
   486→        }
   487→      } else {
   488→        for (let i = 0; i + 1 < args.length; i += 2) {
   489→          const r = args[i];
   490→          const c = args[i + 1];
   491→          pairs.push({ vals: r ? (r.kind === "range" ? r.values : [String(r.v)]) : [], crit: c ? (c.kind === "value" ? String(c.v) : String(c.values[0] ?? "")) : "" });
   492→        }
   493→      }
   494→      const makeTest = (crit: string) => (v: string): boolean => {
   495→        const m = /^(>=|<=|>|<|=|<>)(.*)$/.exec(crit);
   496→        if (m) {
   497→          const target = parseFloat(m[2].replace(/[$,\s]/g, ""));
   498→          const value = parseFloat(String(v).replace(/[$,\s]/g, ""));
   499→          if (!isNaN(target) && !isNaN(value)) {
   500→            switch (m[1]) {
   501→              case ">": return value > target;
   502→              case "<": return value < target;
   503→              case ">=": return value >= target;
   504→              case "<=": return value <= target;
   505→              case "=": return value === target;
   506→              case "<>": return value !== target;
   507→            }
   508→          }
   509→          return m[1] === "=" ? String(v) === m[2] : false;
   510→        }
   511→        return String(v).trim().toLowerCase() === crit.trim().toLowerCase();
   512→      };
   513→      const n = sumVals.length || pairs[0]?.vals.length || 0;
   514→      let out = 0;
   515→      for (let i = 0; i < n; i++) {
   516→        if (pairs.every((p) => { const v = p.vals[i] ?? ""; return v !== "" && makeTest(p.crit)(v); })) {
   517→          if (name === "COUNTIFS") out++;
   518→          else {
   519→            const x = parseFloat(String(sumVals[i] ?? "").replace(/[$,\s]/g, ""));
   520→            if (!isNaN(x)) out += x;
   521→          }
   522→        }
   523→      }
   524→      return out;
   525→    }
   526→    case "XLOOKUP": {
   527→      // XLOOKUP(lookup, lookup_range, return_range, [if_not_found])
   528→      const lookup = scalar(0);
   529→      const lr = args[1];
   530→      const rr = args[2];
   531→      if (!lr || lr.kind !== "range" || !rr || rr.kind !== "range") throw new FormulaError("#VALUE!");
   532→      const lk = typeof lookup === "number" ? lookup : String(lookup).trim().toLowerCase();
   533→      for (let i = 0; i < lr.values.length; i++) {
   534→        const key = lr.values[i];
   535→        const keyN = typeof lookup === "number" ? parseFloat(String(key).replace(/[$,\s]/g, "")) : NaN;
   536→        const match = typeof lookup === "number"
   537→          ? !isNaN(keyN) && keyN === lookup
   538→          : String(key).trim().toLowerCase() === lk;
   539→        if (match) {
   540→          const out = rr.values[i] ?? "";
   541→          const outN = parseFloat(String(out).replace(/[$,\s]/g, ""));
   542→          return isNaN(outN) || out === "" ? out : outN;
   543→        }
   544→      }
   545→      if (args.length > 3) return scalar(3);
   546→      throw new FormulaError("#N/A");
   547→    }
   548→    case "INDEX": {
   549→      const rng = args[0];
   550→      if (!rng || rng.kind !== "range") throw new FormulaError("#VALUE!");
   551→      const row = Math.round(num(scalar(1)));
   552→      const col = args.length > 2 ? Math.round(num(scalar(2))) : 1;
   553→      if (row < 1 || row > rng.h || col < 1 || col > rng.w) throw new FormulaError("#REF!");
   554→      const out = rng.values[(row - 1) * rng.w + (col - 1)] ?? "";
   555→      const outN = parseFloat(String(out).replace(/[$,\s]/g, ""));
   556→      return isNaN(outN) || out === "" ? out : outN;
   557→    }
   558→    case "MATCH": {
   559→      const lookup = scalar(0);
   560→      const rng = args[1];
   561→      if (!rng || rng.kind !== "range") throw new FormulaError("#VALUE!");
   562→      const type = args.length > 2 ? num(scalar(2)) : 1;
   563→      const vals = rng.values;
   564→      const lk = typeof lookup === "number" ? lookup : String(lookup).trim().toLowerCase();
   565→      if (type === 0) {
   566→        for (let i = 0; i < vals.length; i++) {
   567→          const key = vals[i];
   568→          const keyN = typeof lookup === "number" ? parseFloat(String(key).replace(/[$,\s]/g, "")) : NaN;
   569→          const match = typeof lookup === "number" ? !isNaN(keyN) && keyN === lookup : String(key).trim().toLowerCase() === lk;
   570→          if (match) return i + 1;
   571→        }
   572→        throw new FormulaError("#N/A");
   573→      }
   574→      if (type === 1) {
   575→        let best = -1;
   576→        for (let i = 0; i < vals.length; i++) {
   577→          const v = parseFloat(String(vals[i]).replace(/[$,\s]/g, ""));
   578→          if (!isNaN(v) && typeof lk === "number" && v <= lk) best = i; else break;
   579→        }
   580→        if (best < 0) throw new FormulaError("#N/A");
   581→        return best + 1;
   582→      }
   583→      let best = -1;
   584→      for (let i = 0; i < vals.length; i++) {
   585→        const v = parseFloat(String(vals[i]).replace(/[$,\s]/g, ""));
   586→        if (!isNaN(v) && typeof lk === "number" && v >= lk) best = i; else break;
   587→      }
   588→      if (best < 0) throw new FormulaError("#N/A");
   589→      return best + 1;
   590→    }
   591→    case "ROUNDUP": { const d = args.length > 1 ? num(scalar(1)) : 0; const p = Math.pow(10, d); return Math.ceil(Math.abs(num(scalar(0))) * p) / p * Math.sign(num(scalar(0)) || 1); }
   592→    case "ROUNDDOWN": { const d = args.length > 1 ? num(scalar(1)) : 0; const p = Math.pow(10, d); return (Math.floor(Math.abs(num(scalar(0))) * p) / p) * Math.sign(num(scalar(0)) || 1); }
   593→    default: throw new FormulaError(`#${name}?`);
   594→  }
   595→}

   597→function displayValue(ref: string, cells: Record<string, string>): string {
   598→  const raw = cells[ref];
   599→  if (raw === undefined || raw === "") return "";
   600→  if (!raw.startsWith("=")) return raw;
   601→  try {
   602→    const v = evalSheetFormula(raw.slice(1), cells, new Set([ref]));
   603→    if (typeof v === "number") {
   604→      if (Number.isInteger(v)) return v.toLocaleString("en-US");
   605→      return parseFloat(v.toFixed(4)).toLocaleString("en-US", { maximumFractionDigits: 4 });
   606→    }
   607→    return v;
   608→  } catch (e) {
   609→    return e instanceof FormulaError ? e.message : "#ERROR!";
   610→  }
   611→}

   613→function usedRange(cells: Record<string, string>): { rows: number; cols: number } {
   614→  let maxR = 0, maxC = 0;
   615→  for (const ref of Object.keys(cells)) {
   616→    if (!cells[ref]) continue;
   617→    const p = parseRef(ref);
   618→    if (p) { maxR = Math.max(maxR, p[1]); maxC = Math.max(maxC, p[0]); }
   619→  }
   620→  return { rows: Math.min(maxR + 1, ROWS), cols: Math.min(maxC + 1, COLS) };
   621→}
