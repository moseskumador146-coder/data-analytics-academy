    28→const parseRef = (ref: string): [number, number] | null => {
    29→  const m = /^\$?([A-Z])\$?(\d+)$/i.exec(ref.trim());

    34→type Val = string | number;

    37→class FormulaError extends Error {}

    39→function evalSheetFormula(src: string, cells: Record<string, string>, stack: Set<string>): Val {
    40→  let pos = 0;
    41→  const ws = () => { while (pos < src.length && /\s/.test(src[pos])) pos++; };
    42→
    43→  function parseExpr(): Val { return parseCompare(); }
    44→
    45→  function parseCompare(): Val {
    46→    let l = parseConcat();
    47→    ws();
    48→    const ops = ["<=", ">=", "<>", "=", "<", ">"];
    49→    for (const op of ops) {
    50→      if (src.startsWith(op, pos)) {
    51→        pos += op.length;
    52→        const r = parseConcat();
    53→        const ln = typeof l === "number" ? l : parseFloat(String(l));
    54→        const rn = typeof r === "number" ? r : parseFloat(String(r));
    55→        const numeric = !isNaN(ln) && !isNaN(rn);
    56→        switch (op) {
    57→          case "=": return (numeric ? ln === rn : String(l) === String(r)) ? 1 : 0;
    58→          case "<>": return (numeric ? ln !== rn : String(l) !== String(r)) ? 1 : 0;
    59→          case "<": return numeric ? (ln < rn ? 1 : 0) : String(l) < String(r) ? 1 : 0;
    60→          case ">": return numeric ? (ln > rn ? 1 : 0) : String(l) > String(r) ? 1 : 0;
    61→          case "<=": return numeric ? (ln <= rn ? 1 : 0) : String(l) <= String(r) ? 1 : 0;
    62→          case ">=": return numeric ? (ln >= rn ? 1 : 0) : String(l) >= String(r) ? 1 : 0;
    63→        }
    64→      }
    65→    }
    66→    return l;
    67→  }
    68→
    69→  function parseConcat(): Val {
    70→    let l = parseAdd();
    71→    for (;;) {
    72→      ws();
    73→      if (src[pos] === "&") {
    74→        pos++;
    75→        const r = parseAdd();
    76→        l = `${fmtRaw(l)}${fmtRaw(r)}`;
    77→      } else return l;
    78→    }
    79→  }
    80→
    81→  function parseAdd(): Val {
    82→    let l = parseMul();
    83→    for (;;) {
    84→      ws();
    85→      if (src[pos] === "+") { pos++; l = num(l) + num(parseMul()); }
    86→      else if (src[pos] === "-") { pos++; l = num(l) - num(parseMul()); }
    87→      else return l;
    88→    }
    89→  }
    90→  function parseMul(): Val {
    91→    let l = parsePow();
    92→    for (;;) {
    93→      ws();
    94→      if (src[pos] === "*") { pos++; l = num(l) * num(parsePow()); }
    95→      else if (src[pos] === "/") {
    96→        pos++;
    97→        const d = num(parsePow());
    98→        if (d === 0) throw new FormulaError("#DIV/0!");
    99→        l = num(l) / d;
   100→      } else return l;
   101→    }
   102→  }
   103→  function parsePow(): Val {
   104→    let l = parseUnary();
   105→    for (;;) {
   106→      ws();
   107→      if (src[pos] === "^") { pos++; l = Math.pow(num(l), num(parseUnary())); }
   108→      else return l;
   109→    }
   110→  }
   111→  function parseUnary(): Val {
   112→    ws();
   113→    if (src[pos] === "-") { pos++; return -num(parseUnary()); }
   114→    if (src[pos] === "+") { pos++; return num(parseUnary()); }
   115→    return parsePrimary();
   116→  }
   117→
   118→  function parsePrimary(): Val | Arg {
   119→    ws();
   120→    if (src[pos] === "(") {
   121→      pos++;
   122→      const v = parseExpr();
   123→      ws();
   124→      if (src[pos] !== ")") throw new FormulaError("#SYNTAX!");
   125→      pos++;
   126→      return v;
   127→    }
   128→    if (src[pos] === '"') {
   129→      let s = "";
   130→      pos++;
   131→      while (pos < src.length && src[pos] !== '"') s += src[pos++];
   132→      pos++;
   133→      return s;
   134→    }
   135→    const numMatch = /^\d+(\.\d+)?/.exec(src.slice(pos));
   136→    if (numMatch) { pos += numMatch[0].length; return parseFloat(numMatch[0]); }
   137→    const fnMatch = /^([A-Z][A-Z0-9.]*)\(/i.exec(src.slice(pos));
   138→    if (fnMatch) {
   139→      const name = fnMatch[1].toUpperCase();
   140→      pos += fnMatch[0].length;
   141→      const args: Arg[] = [];
   142→      if (src[pos] === ")") pos++;
   143→      else {
   144→        for (;;) {
   145→          args.push(parseArg());
   146→          if (src[pos] === ",") { pos++; continue; }
   147→          if (src[pos] === ")") { pos++; break; }
   148→          throw new FormulaError("#SYNTAX!");
   149→        }
   150→      }
   151→      return callFn(name, args);
   152→    }
   153→    const refMatch = /^\$?[A-Za-z]\$?\d+/.exec(src.slice(pos));
   154→    if (refMatch) {
   155→      const ref = refMatch[0].toUpperCase();
   156→      pos += refMatch[0].length;
   157→      ws();
   158→      if (src[pos] === ":") {
   159→        pos++;
   160→        const ref2Match = /^\$?[A-Za-z]\$?\d+/.exec(src.slice(pos));
   161→        if (!ref2Match) throw new FormulaError("#SYNTAX!");
   162→        pos += ref2Match[0].length;
   163→        return { kind: "range", values: rangeValues(ref, ref2Match[0].toUpperCase(), cells, stack) };
   164→      }
   165→      return { kind: "value", v: cellValue(ref, cells, stack) };
   166→    }
   167→    if (/^TRUE$/i.test(src.slice(pos, pos + 4))) { pos += 4; return 1; }
   168→    if (/^FALSE$/i.test(src.slice(pos, pos + 5))) { pos += 5; return 0; }
   169→    throw new FormulaError("#NAME?");
   170→  }
   171→
   172→  function parseArg(): Arg {
   173→    ws();
   174→    const startRef = /^\$?[A-Za-z]\$?\d+/.exec(src.slice(pos));
   175→    if (startRef) {
   176→      const ref = startRef[0].toUpperCase();
   177→      const after = src.slice(pos + startRef[0].length);
   178→      if (/^\s*:/.test(after)) {
   179→        const m2 = /^\s*:\s*\$?[A-Za-z]\$?\d+/.exec(after);
   180→        if (m2) {
   181→          const ref2 = /^\s*:\s*(\$?[A-Za-z]\$?\d+)/.exec(after)![1].toUpperCase();
   182→          pos += startRef[0].length + /^\s*:\s*\$?[A-Za-z]\$?\d+/.exec(after)![0].length;
   183→          return { kind: "range", values: rangeValues(ref, ref2, cells, stack) };
   184→        }
   185→      }
   186→    }
   187→    const v = parseExpr();
   188→    return { kind: "value", v };
   189→  }
   190→
   191→  const v = parseExpr();
   192→  ws();
   193→  if (pos < src.length) throw new FormulaError("#SYNTAX!");
   194→  return v;
   195→}

   197→function fmtRaw(v: Val): string {
   198→  return typeof v === "number" ? fmtNum(v) : String(v);
   199→}
   200→function num(v: Val): number {
   201→  if (typeof v === "number") return v;
   202→  const n = parseFloat(String(v ?? "").replace(/[$,%\s]/g, ""));
   203→  if (isNaN(n)) throw new FormulaError("#VALUE!");
   204→  return n;
   205→}
   206→function cellValue(ref: string, cells: Record<string, string>, stack: Set<string>): Val {
   207→  if (stack.has(ref)) throw new FormulaError("#CYCLE!");
   208→  const raw = cells[ref] ?? "";
   209→  if (raw === "") return "";
   210→  if (raw.startsWith("=")) {
   211→    stack.add(ref);
   212→    try {
   213→      return evalSheetFormula(raw.slice(1), cells, stack);
   214→    } finally {
   215→      stack.delete(ref);
   216→    }
   217→  }
   218→  const n = parseFloat(raw);
   219→  return isNaN(n) || !/^-?[\d.,$%eE+-]+$/.test(raw) || /[\d]([.,]\d+)*[.,]\D/.test(raw) ? raw : n;
   220→}
   221→function rangeValues(a: string, b: string, cells: Record<string, string>, stack: Set<string>): string[] {
   222→  const [c1, r1] = parseRef(a)!;
   223→  const [c2, r2] = parseRef(b)!;
   224→  const [cMin, cMax] = [Math.min(c1, c2), Math.max(c1, c2)];
   225→  const [rMin, rMax] = [Math.min(r1, r2), Math.max(r1, r2)];
   226→  const out: string[] = [];
   227→  for (let r = rMin; r <= rMax; r++)
   228→    for (let c = cMin; c <= cMax; c++)
   229→      out.push(String(cellValue(refFor(c, r), cells, stack) ?? ""));
   230→  return out;
   231→}

   233→function callFn(name: string, args: Arg[], ): Val {
   234→  const flat = (): string[] => {
   235→    const out: string[] = [];
   236→    for (const a of args) {
   237→      if (a.kind === "range") out.push(...a.values);
   238→      else out.push(String(a.v));
   239→    }
   240→    return out;
   241→  };
   242→  const numeric = (vals: string[]): number[] =>
   243→    vals.map((v) => parseFloat(String(v ?? "").replace(/[$,\s]/g, ""))).filter((n) => !isNaN(n));
   244→  const nonEmpty = (): string[] => flat().filter((v) => v !== "" && v !== "#CYCLE!");
   245→
   246→  switch (name) {
   247→    case "SUM": return numeric(flat()).reduce((s, n) => s + n, 0);
   248→    case "AVERAGE": {
   249→      const n = numeric(flat());
   250→      return n.length ? n.reduce((s, x) => s + x, 0) / n.length : 0;
   251→    }
   252→    case "MEDIAN": {
   253→      const n = numeric(flat()).sort((a, b) => a - b);
   254→      if (!n.length) return 0;
   255→      const mid = Math.floor(n.length / 2);
   256→      return n.length % 2 ? n[mid] : (n[mid - 1] + n[mid]) / 2;
   257→    }
   258→    case "STDEV": {
   259→      const n = numeric(flat());
   260→      if (n.length < 2) return 0;
   261→      const mean = n.reduce((s, x) => s + x, 0) / n.length;
   262→      return Math.sqrt(n.reduce((s, x) => s + (x - mean) ** 2, 0) / (n.length - 1));
   263→    }
   264→    case "COUNT": return numeric(flat()).length;
   265→    case "COUNTA": return nonEmpty().length;
   266→    case "MIN": { const n = numeric(flat()); return n.length ? Math.min(...n) : 0; }
   267→    case "MAX": { const n = numeric(flat()); return n.length ? Math.max(...n) : 0; }
   268→    case "ROUND": {
   269→      const a = args.map((x) => (x.kind === "value" ? x.v : num(x.values[0] ?? "0")));
   270→      const d = a.length > 1 ? num(a[1]) : 0;
   271→      return +num(a[0]).toFixed(d);
   272→    }
   273→    case "ABS": return Math.abs(num(args[0].kind === "value" ? args[0].v : num(args[0].values[0] ?? "0")));
   274→    case "IF": {
   275→      const cond = args[0].kind === "value" ? args[0].v : num(args[0].values[0] ?? "0");
   276→      const yes = args[1]?.kind === "value" ? args[1].v : args[1]?.values[0] ?? "";
   277→      const no = args[2]?.kind === "value" ? args[2].v : args[2]?.values[0] ?? "";
   278→      const truthy = typeof cond === "number" ? cond !== 0 : String(cond).toUpperCase() === "TRUE" || (String(cond) !== "" && String(cond) !== "FALSE");
   279→      return truthy ? yes : no;
   280→    }
   281→    case "AND": return args.every((a) => (typeof a.v === "number" ? a.v !== 0 : String(a.v).toUpperCase() !== "FALSE" && a.v !== "")) ? 1 : 0;
   282→    case "OR": return args.some((a) => (typeof a.v === "number" ? a.v !== 0 : String(a.v).toUpperCase() === "TRUE")) ? 1 : 0;
   283→    case "UPPER": return String(args[0].kind === "value" ? args[0].v : args[0].values[0] ?? "").toUpperCase();
   284→    case "LOWER": return String(args[0].kind === "value" ? args[0].v : args[0].values[0] ?? "").toLowerCase();
   285→    case "TRIM": return String(args[0].kind === "value" ? args[0].v : args[0].values[0] ?? "").trim();
   286→    case "LEN": return String(args[0].kind === "value" ? args[0].v : args[0].values[0] ?? "").length;
   287→    case "SUMIF":
   288→    case "COUNTIF": {
   289→      const rangeArg = args[0];
   290→      const critArg = args[1];
   291→      const crit = critArg.kind === "value" ? critArg.v : critArg.values[0] ?? "";
   292→      const sumRange = name === "SUMIF" && args[2] ? args[2] : null;
   293→      const test = (v: string): boolean => {
   294→        const cs = String(crit);
   295→        const m = /^(>=|<=|>|<|=|<>)(.*)$/.exec(cs);
   296→        if (m) {
   297→          const target = parseFloat(m[2].replace(/[$,\s]/g, ""));
   298→          const value = parseFloat(String(v).replace(/[$,\s]/g, ""));
   299→          if (!isNaN(target) && !isNaN(value)) {
   300→            switch (m[1]) {
   301→              case ">": return value > target;
   302→              case "<": return value < target;
   303→              case ">=": return value >= target;
   304→              case "<=": return value <= target;
   305→              case "=": return value === target;
   306→              case "<>": return value !== target;
   307→            }
   308→          }
   309→          return m[1] === "=" ? String(v) === m[2] : false;
   310→        }
   311→        return String(v).trim().toLowerCase() === cs.trim().toLowerCase();
   312→      };
   313→      if (name === "COUNTIF") {
   314→        const vals = rangeArg.kind === "range" ? rangeArg.values : [String(rangeArg.v)];
   315→        let c = 0;
   316→        for (const v of vals) if (v !== "" && test(v)) c++;
   317→        return c;
   318→      }
   319→      const vals = rangeArg.kind === "range" ? rangeArg.values : [String(rangeArg.v)];
   320→      const sums = sumRange ? (sumRange.kind === "range" ? sumRange.values : [String(sumRange.v)]) : vals;
   321→      let s = 0;
   322→      vals.forEach((v, i) => {
   323→        if (v !== "" && test(v)) {
   324→          const n = parseFloat(String(sums[i] ?? "").replace(/[$,\s]/g, ""));
   325→          if (!isNaN(n)) s += n;
   326→        }
   327→      });
   328→      return s;
   329→    }
   330→    default: throw new FormulaError(`#${name}?`);
   331→  }
   332→}

   334→function displayValue(ref: string, cells: Record<string, string>): string {
   335→  const raw = cells[ref];
   336→  if (raw === undefined || raw === "") return "";
   337→  if (!raw.startsWith("=")) return raw;
   338→  try {
   339→    const v = evalSheetFormula(raw.slice(1), cells, new Set([ref]));
   340→    if (typeof v === "number") {
   341→      if (Number.isInteger(v)) return v.toLocaleString("en-US");
   342→      return parseFloat(v.toFixed(4)).toLocaleString("en-US", { maximumFractionDigits: 4 });
   343→    }
   344→    return v;
   345→  } catch (e) {
   346→    return e instanceof FormulaError ? e.message : "#ERROR!";
   347→  }
   348→}

   350→function usedRange(cells: Record<string, string>): { rows: number; cols: number } {
   351→  let maxR = 0, maxC = 0;
   352→  for (const ref of Object.keys(cells)) {
   353→    if (!cells[ref]) continue;
   354→    const p = parseRef(ref);
   355→    if (p) { maxR = Math.max(maxR, p[1]); maxC = Math.max(maxC, p[0]); }
   356→  }
   357→  return { rows: Math.min(maxR + 1, ROWS), cols: Math.min(maxC + 1, COLS) };
   358→}

   467→    dataRows.sort((a, b) => {
   468→      const av = a[colIdx] ?? "", bv = b[colIdx] ?? "";
   469→      const an = parseFloat(av.replace(/[$,\s]/g, "")), bn = parseFloat(bv.replace(/[$,\s]/g, ""));
   470→      if (!isNaN(an) && !isNaN(bn)) return (an - bn) * dir;
   471→      return av.localeCompare(bv) * dir;
   472→    });
   473→    const next: Record<string, string> = {};
   474→    for (let c = 0; c < C; c++) {
   475→      const h = cells[refFor(c, 0)];
   476→      if (h) next[refFor(c, 0)] = h;
   477→    }
   478→    dataRows.forEach((row, ri) => {
   479→      row.forEach((v, ci) => { if (v !== "") next[refFor(ci, ri + 1)] = v; });
   480→    });
   481→    setCells(next);
   482→  };
   483→
   484→  const dedupeRows = () => {
   485→    const { rows: R, cols: C } = usedRange(cells);
   486→    const seen = new Set<string>();
   487→    const next: Record<string, string> = {};
   488→    for (let c = 0; c < C; c++) { const h = cells[refFor(c, 0)]; if (h) next[refFor(c, 0)] = h; }
   489→    let kept = 0, removed = 0;
   490→    for (let r = 1; r < R; r++) {
   491→      const key = Array.from({ length: C }, (_, c) => displayValue(refFor(c, r), cells)).join("\u0001");
   492→      if (key.replace(/[\u0001]+$/g, "") === "") continue;
   493→      if (seen.has(key)) { removed++; continue; }
   494→      seen.add(key);
   495→      for (let c = 0; c < C; c++) { const v = cells[refFor(c, r)]; if (v) next[refFor(c, kept + 1)] = v; }
   496→      kept++;
   497→    }
   498→    setCells(next);
   499→    if (removed > 0) gridRef.current?.focus();
   500→  };
   501→
   502→  const colStats = React.useMemo(() => {
   503→    const stats: { n: number; sum: number; mean: number; median: number; std: number; min: number; max: number } | null = (() => {
   504→      const vals: number[] = [];
   505→      for (let r = 0; r < usedR; r++) {
   506→        const v = parseFloat(String(displayValue(refFor(selectedCol, r), cells)).replace(/[$,\s]/g, ""));
   507→        if (!isNaN(v) && String(displayValue(refFor(selectedCol, r), cells)) !== "") vals.push(v);
   508→      }
   509→      if (vals.length < 1) return null;
   510→      const sorted = [...vals].sort((a, b) => a - b);
   511→      const sum = vals.reduce((s, x) => s + x, 0);
   512→      const mean = sum / vals.length;
   513→      const mid = Math.floor(sorted.length / 2);
   514→      const median = sorted.length % 2 ? sorted[mid] : (sorted[mid - 1] + sorted[mid]) / 2;
   515→      const std = sorted.length > 1 ? Math.sqrt(vals.reduce((s, x) => s + (x - mean) ** 2, 0) / (vals.length - 1)) : 0;
   516→      return { n: vals.length, sum, mean, median, std, min: sorted[0], max: sorted[sorted.length - 1] };
   517→    })();
   518→    return stats;
   519→  }, [cells, selectedCol, usedR]);
   520→
   521→  const transformCol = (mode: "trim" | "upper" | "lower" | "title") => {
   522→    const next = { ...cells };
   523→    for (let r = 0; r < usedR; r++) {
   524→      const ref = refFor(selectedCol, r);
   525→      const v = next[ref];
   526→      if (v === undefined || v.startsWith("=")) continue;
   527→      if (mode === "trim") next[ref] = v.trim();
   528→      if (mode === "upper") next[ref] = v.toUpperCase();
   529→      if (mode === "lower") next[ref] = v.toLowerCase();
   530→      if (mode === "title") next[ref] = v.toLowerCase().replace(/\b\w/g, (ch) => ch.toUpperCase());
   531→    }
   532→    setCells(next);
   533→  };
   534→
   535→  const activeRaw = cells[active] ?? "";
   536→  const [ac, ar] = parseRef(active)!;
   537→
   538→  return (
   539→    <div className="space-y-4">
   540→      <ToolHeader
   541→        icon={<FileSpreadsheet className="h-5 w-5 text-emerald-400" />}
   542→        title="Excel Studio"
   543→        subtitle="Full spreadsheet with formula engine — SUM to SUMIF, stats, sort, dedupe, CSV in/out"
   544→        actions={
   545→          <>
   546→            <Select onValueChange={(v) => loadDataset(v)}>
   547→              <SelectTrigger className="w-[190px] border-white/15 bg-zinc-900 text-sm"><Database className="mr-1 h-4 w-4 text-emerald-400" /><SelectValue placeholder="Load sample data…" /></SelectTrigger>
   548→              <SelectContent className="border-white/10 bg-zinc-900">
   549→                {getAllDatasets().map((d) => <SelectItem key={d.id} value={d.id}>{d.name}</SelectItem>)}
   550→                <SelectItem value="messy_sales">Retail Sales H2-2024 (Messy)</SelectItem>
   551→              </SelectContent>
   552→            </Select>
   553→            <input ref={fileRef} type="file" accept=".csv" className="hidden" onChange={(e) => e.target.files?.[0] && importCSV(e.target.files[0])} />
   554→            <Button variant="outline" size="sm" className="border-white/15 bg-transparent hover:bg-white/10" onClick={() => fileRef.current?.click()}><Upload className="h-4 w-4" /> Import CSV</Button>
   555→            <Button variant="outline" size="sm" className="border-white/15 bg-transparent hover:bg-white/10" onClick={exportCSV}><Download className="h-4 w-4" /> Export CSV</Button>
   556→          </>
   557→        }
   558→      />
   559→
   560→      {/* formula bar */}
   561→      <div className="flex items-center gap-2">
   562→        <div className="flex h-9 w-20 shrink-0 items-center justify-center rounded-lg border border-white/15 bg-black/40 font-mono text-sm font-bold text-emerald-300">{active}</div>
   563→        <div className="flex h-9 flex-1 items-center rounded-lg border border-white/15 bg-black/40 px-3 font-mono text-sm text-zinc-200">
   564→          <Sigma className="mr-2 h-3.5 w-3.5 shrink-0 text-zinc-500" />
   565→          <input
   566→            className="w-full bg-transparent outline-none placeholder:text-zinc-600"
   567→            placeholder="Type a value or =SUM(A1:A50)  ·  =IF(A2>100,&quot;Big&quot;,&quot;Small&quot;)"
   568→            value={editVal ?? activeRaw}
   569→            onChange={(e) => setEditVal(e.target.value)}
   570→            onKeyDown={(e) => {
   571→              if (e.key === "Enter") { commit(); gridRef.current?.focus(); }
   572→              if (e.key === "Escape") setEditVal(null);
   573→            }}
   574→            onFocus={() => { if (editVal === null) setEditVal(activeRaw); }}
   575→          />
   576→        </div>
   577→      </div>
   578→
   579→      {/* toolbar */}
   580→      <div className={`${PANEL} flex flex-wrap items-center gap-1.5 px-2 py-1.5 text-xs`}>
   581→        <span className="px-1 text-zinc-500">Sort col {colName(selectedCol)}:</span>
   582→        <Button variant="ghost" size="sm" className="h-7 px-2 text-xs" onClick={() => sortRows(selectedCol, 1)}>A→Z</Button>
   583→        <Button variant="ghost" size="sm" className="h-7 px-2 text-xs" onClick={() => sortRows(selectedCol, -1)}>Z→A</Button>
   584→        <span className="mx-1 h-4 w-px bg-white/10" />
   585→        <Button variant="ghost" size="sm" className="h-7 px-2 text-xs" onClick={dedupeRows}><Eraser className="mr-1 h-3 w-3" />Remove duplicate rows</Button>
   586→        <span className="mx-1 h-4 w-px bg-white/10" />
   587→        <span className="px-1 text-zinc-500">Text col {colName(selectedCol)}:</span>
   588→        {(["trim", "upper", "lower", "title"] as const).map((m) => (

   590→        ))}
   591→        <span className="mx-1 h-4 w-px bg-white/10" />

   593→        <span className="ml-auto flex items-center gap-1.5">
   594→          <Input value={sheetName} onChange={(e) => setSheetName(e.target.value)} placeholder="Sheet name" className="h-7 w-28 border-white/15 bg-black/30 text-xs" />
   595→          <Button variant="outline" size="sm" className="h-7 border-white/15 px-2 text-xs" onClick={() => sheetName && saveSheet(sheetName, cells)}><Save className="h-3 w-3" /> Save</Button>
   596→          {Object.keys(sheets).length > 0 && (
   597→            <Select onValueChange={(v) => { setCells(sheets[v].cells); setSheetName(v); }}>
   598→              <SelectTrigger className="h-7 w-[130px] border-white/15 bg-zinc-900 text-xs"><FolderOpen className="mr-1 h-3 w-3 text-emerald-400" /><SelectValue placeholder="Open saved…" /></SelectTrigger>
   599→              <SelectContent className="border-white/10 bg-zinc-900">
   600→                {Object.values(sheets).map((s) => (
   601→                  <SelectItem key={s.name} value={s.name}>
   602→                    <span className="flex items-center justify-between gap-2"><span>{s.name}</span><span className="text-[10px] text-zinc-500">{new Date(s.savedAt).toLocaleDateString()}</span></span>
   603→                  </SelectItem>
   604→                ))}
   605→              </SelectContent>
   606→            </Select>
   607→          )}
   608→        </span>
   609→      </div>
   610→
   611→      {/* grid + stats */}
   612→      <div className="flex flex-col gap-4 lg:flex-row">
   613→        <div
   614→          ref={gridRef}
   615→          tabIndex={0}
   616→          onKeyDown={onKey}
   617→          className={`${PANEL} flex-1 overflow-auto outline-none scrollbar-thin`} style={{ maxHeight: "62vh" }}
   618→        >
   619→          <table className="border-collapse text-[13px]" style={{ minWidth: viewCols * 96 }}>

   621→              <tr className="sticky top-0 z-10 bg-zinc-900">
   622→                <th className="sticky left-0 z-20 w-10 border-b border-r border-white/10 bg-zinc-900 px-1 py-1.5 text-[10px] font-medium text-zinc-500" />

   631→                    {colName(c)}
   632→                  </th>

   634→              </tr>
   635→            </thead>

   637→              {Array.from({ length: viewRows }, (_, r) => (

   639→                  <td className={`sticky left-0 z-10 w-10 border-b border-r border-white/10 bg-zinc-900 px-1 py-1 text-center text-[10px] font-medium ${active.endsWith(String(r + 1)) ? "text-emerald-300" : "text-zinc-500"}`}>{r + 1}</td>
   640→                  {Array.from({ length: viewCols }, (_, c) => {

   719→                    <button className="text-zinc-500 hover:text-red-400" onClick={() => deleteSheet(s.name)}><Trash2 className="h-3 w-3" /></button>
