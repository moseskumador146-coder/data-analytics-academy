   121→type RunResult = { columns: string[]; rows: (string | number | null)[][]; ms: number };
   122→
   123→/** Convert any sample dataset into a queryable SQL table (types inferred, blanks → NULL) */
   124→function datasetToSqlTable(ds: Dataset): SqlTable {
   125→  const columns = ds.columns.map((c) => {
   126→    let type = "TEXT";
   127→    if (c.type === "date") type = "DATE";
   128→    else if (c.type === "number" || c.type === "currency") {
   129→      const allNum = ds.rows.every((r) => {
   130→        const v = r[c.key];
   131→        if (v === null || v === "" || v === undefined) return true;
   132→        return typeof v === "number" || !isNaN(parseFloat(String(v).replace(/[$,\s]/g, "")));
   133→      });
   134→      type = allNum ? "REAL" : "TEXT";
   135→    }
   136→    return { name: c.key, type };
   137→  });
   138→  const rows: Row[] = ds.rows.slice(0, 4000).map((r) => {
   139→    const out: Row = {};
   140→    for (const c of ds.columns) {
   141→      const v = r[c.key];
   142→      if (v === "" || v === undefined || v === null) out[c.key] = null;
   143→      else if (typeof v === "string") {
   144→        const n = parseFloat(v.replace(/[$,\s]/g, ""));
   145→        out[c.key] = !isNaN(n) && /^[\d.,$\s+-]+$/.test(v) ? n : v;
   146→      } else out[c.key] = v;
   147→    }
   148→    return out;
   149→  });
   150→  return {
   151→    name: ds.id,
   152→    description: `${ds.name} — imported sample file (first ${rows.length.toLocaleString()} rows; blanks became NULL, $-text became numbers).`,
   153→    columns,
   154→    rows,
   155→  };
   156→}
